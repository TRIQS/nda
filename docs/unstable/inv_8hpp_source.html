72 template <MemoryMatrix M>
73 requires(get_algebra<M> == 'M' and mem::on_host<M>)
74 void inverse2_in_place(M &&m) { // NOLINT (temporary views are allowed here)
75 // calculate the adjoint of the matrix
76 std::swap(m(0, 0), m(1, 1));
77
78 // calculate the inverse determinant of the matrix
79 auto det = (m(0, 0) * m(1, 1) - m(0, 1) * m(1, 0));
80 if (det == 0.0) NDA_RUNTIME_ERROR << "Error in nda::inverse2_in_place: Matrix is not invertible";
81 auto detinv = 1.0 / det;
82
83 // multiply the adjoint by the inverse determinant
84 m(0, 0) *= +detinv;
85 m(1, 1) *= +detinv;
86 m(1, 0) *= -detinv;
87 m(0, 1) *= -detinv;
88 }

98 template <MemoryMatrix M>
99 requires(get_algebra<M> == 'M' and mem::on_host<M>)
100 void inverse3_in_place(M &&m) { // NOLINT (temporary views are allowed here)
101 // calculate the cofactors of the matrix
102 auto b00 = +m(1, 1) * m(2, 2) - m(1, 2) * m(2, 1);
103 auto b10 = -m(1, 0) * m(2, 2) + m(1, 2) * m(2, 0);
104 auto b20 = +m(1, 0) * m(2, 1) - m(1, 1) * m(2, 0);
105 auto b01 = -m(0, 1) * m(2, 2) + m(0, 2) * m(2, 1);
106 auto b11 = +m(0, 0) * m(2, 2) - m(0, 2) * m(2, 0);
107 auto b21 = -m(0, 0) * m(2, 1) + m(0, 1) * m(2, 0);
108 auto b02 = +m(0, 1) * m(1, 2) - m(0, 2) * m(1, 1);
109 auto b12 = -m(0, 0) * m(1, 2) + m(0, 2) * m(1, 0);
110 auto b22 = +m(0, 0) * m(1, 1) - m(0, 1) * m(1, 0);
111
112 // calculate the inverse determinant of the matrix
113 auto det = m(0, 0) * b00 + m(0, 1) * b10 + m(0, 2) * b20;
114 if (det == 0.0) NDA_RUNTIME_ERROR << "Error in nda::inverse3_in_place: Matrix is not invertible";
115 auto detinv = 1.0 / det;
116
117 // fill the matrix by multiplying the cofactors by the inverse determinant
118 m(0, 0) = detinv * b00;
119 m(0, 1) = detinv * b01;
120 m(0, 2) = detinv * b02;
121 m(1, 0) = detinv * b10;
122 m(1, 1) = detinv * b11;
123 m(1, 2) = detinv * b12;
124 m(2, 0) = detinv * b20;
125 m(2, 1) = detinv * b21;
126 m(2, 2) = detinv * b22;
127 }

142 template <MemoryMatrix M>
143 requires(get_algebra<M> == 'M')
144 void inverse_in_place(M &&m) { // NOLINT (temporary views are allowed here)
145 EXPECTS(is_matrix_square(m, true));
146
147 // nothing to do if the matrix/view is empty
148 if (m.empty()) return;
149
150 // use optimized routines for small matrices
151 if constexpr (mem::on_host<M>) {
152 if (m.shape()[0] == 1) {
153 inverse1_in_place(m);
154 return;
155 }
156
157 if (m.shape()[0] == 2) {
158 inverse2_in_place(m);
159 return;
160 }
161
162 if (m.shape()[0] == 3) {
163 inverse3_in_place(m);
164 return;
165 }
166 }
167
168 // use getrf and getri from lapack for larger matrices
169 array<int, 1> ipiv(m.extent(0));
170 int info = lapack::getrf(m, ipiv); // it is ok to be in C order
171 if (info != 0) NDA_RUNTIME_ERROR << "Error in nda::inverse_in_place: Matrix is not invertible: info = " << info;
172 info = lapack::getri(m, ipiv);
173 if (info != 0) NDA_RUNTIME_ERROR << "Error in nda::inverse_in_place: Matrix is not invertible: info = " << info;
174 }
