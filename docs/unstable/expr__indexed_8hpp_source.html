44 // Extract a subset of indices from a container using a range.
45 template <typename Container>
46 std::vector<long> extract_indices(Container const &c, range const &rg) {
47 auto len = static_cast<long>(c.size());
48 auto last = (rg.last() == -1 && rg.step() > 0) ? len : rg.last();
49 range actual_range(rg.first(), last, rg.step());
50 std::vector<long> result;
51 result.reserve(actual_range.size());
52 for (long i : actual_range) { result.push_back(c[static_cast<size_t>(i)]); }
53 return result;
54 }

110 // Apply a function to the nth index container at runtime.
111 template <typename F, size_t... Is>
112 [[nodiscard]] auto apply_to_nth(int n, F &&f, std::index_sequence<Is...>) const {
113 using R = std::invoke_result_t<F, std::tuple_element_t<0, decltype(idx)> const &>;
114 R result{};
115 (void)((static_cast<int>(Is) == n ? (result = f(std::get<Is>(idx)), true) : false) || ...);
116 return result;
117 }

119 // Find which index container (if any) handles dimension d. Returns -1 if none.
120 [[nodiscard]] constexpr int find_container_for_dim(int d) const {
121 for (int i = 0; i < n_idx_containers; ++i) {
122 if (indexed_dims[i] == d) return i;
123 }
124 return -1;
125 }

127 // Resolve an index for a given dimension: map through container or pass through.
128 [[nodiscard]] long resolve_index(int dim, long input_idx) const {
129 int c = find_container_for_dim(dim);
130 if (c < 0) return input_idx;
131 return apply_to_nth(c, [i = input_idx](auto const &vec) { return static_cast<long>(vec[static_cast<size_t>(i)]); },
132 std::make_index_sequence<n_idx_containers>{});
133 }

135 // Element access implementation - const version.
136 template <size_t... Is, typename... Args>
137 [[nodiscard]] decltype(auto) call_impl(std::index_sequence<Is...>, Args... args) const {
138 std::array<long, rank> a = {static_cast<long>(args)...};
139 return v(resolve_index(static_cast<int>(Is), a[Is])...);
140 }

143 template <size_t... Is, typename... Args>

248 // Main slicing implementation - returns expr_indexed with same container types.
249 template <typename... Args>
250 [[nodiscard]] auto slice_impl(Args const &...args) const {
251 auto args_tuple = std::make_tuple(args...);
252
253 // Build dimension mapping (old -> new) based on which args are not long
254 constexpr auto dim_kept = compute_dim_kept<Args...>();
255 std::array<int, rank> old_to_new{};
256 int new_idx = 0;
257 for (int d = 0; d < rank; ++d) { old_to_new[d] = dim_kept[d] ? new_idx++ : -1; }
258
259 // Build the sliced underlying view
260 auto sliced_view = build_sliced_view(args_tuple);
261
262 // Process each index container to build new containers
263 auto new_containers_tuple = [&]<size_t... Cs>(std::index_sequence<Cs...>) {
264 return std::make_tuple(slice_single_container<Cs>(args_tuple)...);
265 }(std::make_index_sequence<n_idx_containers>{});
266
267 // Map indexed dimensions to new indices
268 std::array<int, n_idx_containers> new_indexed_dims{};
269 for (int c = 0; c < n_idx_containers; ++c) {
270 new_indexed_dims[c] = old_to_new[indexed_dims[c]];
271 }
272
273 return expr_indexed<decltype(sliced_view), IndexContainers...>{std::move(sliced_view), std::move(new_containers_tuple),
274 new_indexed_dims};
275 }

277 // Build sliced view: for indexed dims pass full range, for non-indexed dims pass the slice arg.
278 template <typename ArgTuple, size_t... Dims>
279 [[nodiscard]] auto build_sliced_view_impl(ArgTuple const &args_tuple, std::index_sequence<Dims...>) const {
280 // Convert all arguments to range for uniform handling.
281 auto to_range = [&]<size_t D>() -> range {
282 if (find_container_for_dim(static_cast<int>(D)) >= 0) {
283 return range(0, v.shape()[D]);
284 }
285 using ArgT = std::remove_cvref_t<std::tuple_element_t<D, ArgTuple>>;
286 auto const &arg = std::get<D>(args_tuple);
287 if constexpr (std::is_same_v<ArgT, range>) {
288 return arg;
289 } else if constexpr (std::is_same_v<ArgT, range::all_t>) {
290 return range(0, v.shape()[D]);
291 } else {
292 return range(static_cast<long>(arg), static_cast<long>(arg) + 1);
293 }
294 };
295 return v(to_range.template operator()<Dims>()...);
296 }

303 // Slice container C using the argument at dimension D.
304 template <size_t C, size_t D, typename ArgTuple>
305 [[nodiscard]] std::vector<long> slice_container_at_dim(ArgTuple const &args_tuple) const {
306 auto const &container = std::get<C>(idx);
307 using ArgT = std::remove_cvref_t<std::tuple_element_t<D, ArgTuple>>;
308 if constexpr (std::is_same_v<ArgT, range>) {
309 return detail::extract_indices(container, std::get<D>(args_tuple));
310 } else {
311 return {container.begin(), container.end()};
312 }
313 }

316 template <size_t C, typename ArgTuple, size_t... Dims>
317 [[nodiscard]] std::vector<long> slice_single_container_impl(ArgTuple const &args_tuple, std::index_sequence<Dims...>) const {

358 // Compute which argument positions contain IndexContainers.
359 template <typename... Ts>
360 constexpr auto compute_indexed_dims() {
361 constexpr int n_idx = (static_cast<int>(IndexContainer<std::remove_cvref_t<Ts>>) + ...);
362 constexpr std::array<bool, sizeof...(Ts)> mask = {IndexContainer<std::remove_cvref_t<Ts>>...};
363 std::array<int, n_idx> result{};
364 int idx = 0;
365 for (int i = 0; i < static_cast<int>(sizeof...(Ts)); ++i) {
366 if (mask[i]) result[idx++] = i;
367 }
368 return result;
369 }
