57 template <Vector WR, Vector WI>
58 requires(mem::have_host_compatible_addr_space<WR, WI> and std::same_as<double, get_value_t<WR>> and have_same_value_type_v<WR, WI>)
59 auto get_geev_eigenvalues(const WR &wr, const WI &wi) {
60 // check the dimensions of the input arrays/views
61 auto const n = wr.size();
62 EXPECTS(n == wi.size());
63
64 // generate eigenvalues
65 auto lambda = array<std::complex<double>, 1>(n);
66 for (long i = 0; i < n; ++i) lambda(i) = std::complex<double>(wr(i), wi(i));
67 return lambda;
68 }

95 template <Vector WI, Matrix VA>
96 requires(mem::have_host_compatible_addr_space<WI, VA> and std::same_as<double, get_value_t<WI>> and have_same_value_type_v<WI, VA>)
97 auto get_geev_eigenvectors(const WI &wi, const VA &va) {
98 using namespace std::complex_literals;
99 static_assert(nda::blas::has_F_layout<VA>, "Error in nda::linalg::get_geev_eigenvectors: VA must have Fortran layout");
100
101 // check the dimensions of the input arrays/views
102 auto const n = wi.size();
103 EXPECTS(va.shape() == (std::array<long, 2>{n, n}));
104
105 // unpack eigenvectors
106 auto X = matrix<std::complex<double>, F_layout>(n, n);
107 long j = 0;
108 while (j < n) {
109 if (wi(j) > 0.0) {
110 // complex conjugate eigenvalue pair --> we need to unpack the eigenvectors
111 for (long i = 0; i < n; ++i) {
112 X(i, j) = std::complex<double>{va(i, j), va(i, j + 1)};
113 X(i, j + 1) = std::complex<double>{va(i, j), -va(i, j + 1)};
114 }
115 j += 2;
116 } else {
117 // real eigenvalue --> eigenvector is purely real
118 X(nda::range::all, j) = va(nda::range::all, j);
119 ++j;
120 }
121 }
122
123 return X;
124 }

128 // Implementation for complex matrices - straightforward call to geev.
129 template <MemoryMatrix A>
130 requires(is_complex_v<get_value_t<A>>)
131 auto eig_impl(A &&a, char jobvl, char jobvr) { // NOLINT (temporary views are allowed here)
132 using arr_t = array<get_value_t<A>, 1>;
133 using mat_t = matrix<get_value_t<A>, F_layout>;
134 auto const n = a.extent(0);
135
136 // early return if the matrix is empty
137 if (a.empty()) return std::make_tuple(arr_t{}, mat_t{}, mat_t{});
138
139 // allocate outputs
140 auto lambda = arr_t(n);
141 auto U = mat_t(jobvl == 'V' ? n : 0, jobvl == 'V' ? n : 0);
142 auto V = mat_t(jobvr == 'V' ? n : 0, jobvr == 'V' ? n : 0);
143
144 // make the call to geev
145 int info = nda::lapack::geev(a, lambda, U, V, jobvl, jobvr);
146 if (info != 0) NDA_RUNTIME_ERROR << "Error in nda::linalg::detail::eig_impl: geev routine failed: info = " << info;
147
148 return std::make_tuple(std::move(lambda), std::move(U), std::move(V));
149 }

151 // Implementation for real matrices.
152 template <MemoryMatrix A>
153 requires(std::same_as<double, get_value_t<A>>)
154 auto eig_impl(A &&a, char jobvl, char jobvr) { // NOLINT (temporary views are allowed here)
155 using arr_t = array<std::complex<double>, 1>;
156 using mat_t = matrix<std::complex<double>, F_layout>;
157 auto const n = a.extent(0);
158
159 // early return if the matrix is empty
160 if (a.empty()) return std::make_tuple(arr_t{}, mat_t{}, mat_t{});
161
162 // allocate outputs
163 auto wr = array<double, 1>(n);
164 auto wi = array<double, 1>(n);
165 auto vl = matrix<double, F_layout>(jobvl == 'V' ? n : 0, jobvl == 'V' ? n : 0);
166 auto vr = matrix<double, F_layout>(jobvr == 'V' ? n : 0, jobvr == 'V' ? n : 0);
167
168 // make the call to geev
169 int info = nda::lapack::geev(a, wr, wi, vl, vr, jobvl, jobvr);
170 if (info != 0) NDA_RUNTIME_ERROR << "Error in nda::linalg::detail::eig_impl: geev routine failed: info = " << info;
171
172 // get eigenvalues and eigenvectors from geev output
173 auto lambda = get_geev_eigenvalues(wr, wi);
174 auto U = (jobvl == 'V') ? get_geev_eigenvectors(wi, vl) : mat_t{};
175 auto V = (jobvr == 'V') ? get_geev_eigenvectors(wi, vr) : mat_t{};
176
177 return std::make_tuple(std::move(lambda), std::move(U), std::move(V));
178 }
