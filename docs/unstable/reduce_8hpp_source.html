45 // Helper function that (all)reduces arrays/views.
46 template <typename A1, typename A2>
47 requires(is_regular_or_view_v<A1> && is_regular_or_view_v<A2>)
48 void mpi_reduce_impl(A1 const &a_in, A2 &&a_out, mpi::communicator comm = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) { // NOLINT
49 // check the shape of the input arrays/views
50 EXPECTS_WITH_MESSAGE(have_mpi_equal_shapes(a_in, comm), "Error in nda::detail::mpi_reduce_impl: Shapes of arrays/views must be equal")
51
52 // simply copy if there is no active MPI environment or if the communicator size is < 2
53 if (not mpi::has_env || comm.size() < 2) {
54 a_out = a_in;
55 return;
56 }
57
58 // reduce the arrays/views
59 using value_type = typename std::decay_t<A1>::value_type;
60 if constexpr (not mpi::has_mpi_type<value_type>) {
61 // arrays/views of non-MPI types are reduced element-wise
62 a_out = nda::map([&](auto const &x) { return mpi::reduce(x, comm, root, all, op); })(a_in);
63 } else {
64 // for MPI-types we have to perform some checks on the input and ouput arrays/views
65 check_layout_mpi_compatible(a_in, "detail::mpi_reduce_impl");
66 if ((comm.rank() == root) || all) {
67 check_layout_mpi_compatible(a_out, "detail::mpi_reduce_impl");
68 nda::resize_or_check_if_view(a_out, a_in.shape());
69 if (std::abs(a_out.data() - a_in.data()) < a_in.size()) NDA_RUNTIME_ERROR << "Error in nda::detail::mpi_reduce_impl: Overlapping arrays";
70 }
71
72 // reduce the data
73 auto a_out_span = std::span{a_out.data(), static_cast<std::size_t>(a_out.size())};
74 auto a_in_span = std::span{a_in.data(), static_cast<std::size_t>(a_in.size())};
75 mpi::reduce_range(a_in_span, a_out_span, comm, root, all, op);
76 }
77 }

79 // Helper function that (all)reduces arrays/views in-place.
80 template <typename A>
81 requires(is_regular_or_view_v<A>)
82 void mpi_reduce_in_place_impl(A &&a_out, mpi::communicator comm = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) { // NOLINT
83 // check the shape of the input arrays/views
84 EXPECTS_WITH_MESSAGE(have_mpi_equal_shapes(a_out, comm), "Error in nda::detail::mpi_reduce_in_place_impl: Shapes of arrays/views must be equal")
85
86 // do nothing if there is no active MPI environment or if the communicator size is < 2
87 if (not mpi::has_env || comm.size() < 2) { return; }
88
89 // reduce the arrays/views
90 using value_type = typename std::decay_t<A>::value_type;
91 if constexpr (not mpi::has_mpi_type<value_type>) {
92 // arrays/views of non-MPI types are reduced element-wise
93 nda::for_each(a_out.shape(), [&](auto... args) { mpi::reduce_in_place(a_out(args...), comm, root, all, op); });
94 } else {
95 // for MPI-types we have to perform some checks on the input/ouput arrays/views
96 check_layout_mpi_compatible(a_out, "detail::mpi_reduce_in_place_impl");
97
98 // reduce the data
99 auto a_out_span = std::span{a_out.data(), static_cast<std::size_t>(a_out.size())};
100 mpi::reduce_in_place_range(a_out_span, comm, root, all, op);
101 }
102 }

120template <nda::Array A>
121struct mpi::lazy<mpi::tag::reduce, A> {
123 using value_type = typename std::decay_t<A>::value_type;
124
126 using stored_type = A;
127
129 stored_type rhs;
130
132 mpi::communicator comm;
133
135 const int root{0}; // NOLINT (const is fine here)
136
138 const bool all{false}; // NOLINT (const is fine here)
139
141 const MPI_Op op{MPI_SUM}; // NOLINT (const is fine here)
142
153 [[nodiscard]] auto shape() const {
154 if ((comm.rank() == root) || all) return rhs.shape();
155 return std::array<long, std::remove_cvref_t<stored_type>::rank>{};
156 }
157
176 template <nda::Array T>
177 void invoke(T &&target) const { // NOLINT (temporary views are allowed here)
178 if (target.data() == rhs.data()) {
179 nda::detail::mpi_reduce_in_place_impl(target, comm, root, all, op);
180 } else {
181 nda::detail::mpi_reduce_impl(rhs, target, comm, root, all, op);
182 }
183 }
184};
