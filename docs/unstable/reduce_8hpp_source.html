78 template <typename A1, typename A2>
79 requires(is_regular_or_view_v<A1> && is_regular_or_view_v<A2>)
80 void mpi_reduce_into(A1 const &a_in, A2 &&a_out, mpi::communicator comm = {}, int root = 0, bool all = false, MPI_Op op = MPI_SUM) { // NOLINT
81 // check the shape of the input arrays/views
82 EXPECTS_WITH_MESSAGE(detail::have_mpi_equal_shapes(a_in, comm), "Error in nda::mpi_reduce_into: Shapes of arrays/views must be equal");
83
84 // resize or check the output array/view on receiving ranks
85 bool const receives = (all || (comm.rank() == root));
86 if (receives) resize_or_check_if_view(a_out, a_in.shape());
87
88 // call mpi::reduce_range with a span if the input and output arrays/views are contiguous with positive strides
89 // (on non-receiving ranks, the output should always be contiguous since it is not used)
90 if (a_in.is_contiguous() and a_in.has_positive_strides()) {
91 auto a_in_span = std::span{a_in.data(), static_cast<std::size_t>(a_in.size())};
92 if ((a_out.is_contiguous() and a_out.has_positive_strides()) || !receives) {
93 auto a_out_span = std::span{a_out.data(), static_cast<std::size_t>(a_out.size())};
94 mpi::reduce_range(a_in_span, a_out_span, comm, root, all, op);
95 } else {
96 mpi::reduce_range(a_in_span, a_out, comm, root, all, op);
97 }
98 } else {
99 if ((a_out.is_contiguous() and a_out.has_positive_strides()) || !receives) {
100 auto a_out_span = std::span{a_out.data(), static_cast<std::size_t>(a_out.size())};
101 mpi::reduce_range(a_in, a_out_span, comm, root, all, op);
102 } else {
103 mpi::reduce_range(a_in, a_out, comm, root, all, op);
104 }
105 }
106 }
