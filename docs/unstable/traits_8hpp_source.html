221 enum class layout_prop_e : uint64_t {
222 none = 0x0,
223 strided_1d = 0x1,
224 smallest_stride_is_one = 0x2,
225 contiguous = strided_1d | smallest_stride_is_one
226 };
227
236 inline constexpr bool layout_property_compatible(layout_prop_e from, layout_prop_e to) {
237 if (from == layout_prop_e::contiguous) return true;
238 return ((to == layout_prop_e::none) or (to == from));
239 }
240
248 constexpr layout_prop_e operator|(layout_prop_e lhs, layout_prop_e rhs) { return layout_prop_e(uint64_t(lhs) | uint64_t(rhs)); }
249
257 constexpr layout_prop_e operator&(layout_prop_e lhs, layout_prop_e rhs) { return layout_prop_e{uint64_t(lhs) & uint64_t(rhs)}; }
