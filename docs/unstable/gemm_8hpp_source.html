47 template <Matrix A, Matrix B, MemoryMatrix C>
48 void gemm_generic(typename A::value_type alpha, A const &a, B const &b, typename A::value_type beta,
49 C &&c) { // NOLINT (temporary views are allowed here)
50 EXPECTS(a.extent(1) == b.extent(0));
51 EXPECTS(a.extent(0) == c.extent(0));
52 EXPECTS(b.extent(1) == c.extent(1));
53 for (int i = 0; i < a.extent(0); ++i) {
54 for (int j = 0; j < b.extent(1); ++j) {
55 c(i, j) = beta * c(i, j);
56 for (int k = 0; k < a.extent(1); ++k) c(i, j) += alpha * a(i, k) * b(k, j);
57 }
58 }
59 }

87 template <Matrix A, Matrix B, MemoryMatrix C>
88 requires((MemoryMatrix<A> or is_conj_array_expr<A>) and (MemoryMatrix<B> or is_conj_array_expr<B>)
89 and have_same_value_type_v<A, B, C> and is_blas_lapack_v<get_value_t<A>>)
90 void gemm(get_value_t<A> alpha, A const &a, B const &b, get_value_t<A> beta, C &&c) {
91 // get underlying matrix in case it is given as a lazy expression
92 auto to_mat = []<typename Z>(Z const &z) -> auto & {
93 if constexpr (is_conj_array_expr<Z>)
94 return std::get<0>(z.a);
95 else
96 return z;
97 };
98 auto &mat_a = to_mat(a);
99 auto &mat_b = to_mat(b);
100
101 // compile-time checks
102 using mat_a_type = decltype(mat_a);
103 using mat_b_type = decltype(mat_b);
104 static_assert(mem::have_compatible_addr_space<mat_a_type, mat_b_type, C>, "Error in nda::blas::gemm: Incompatible memory address spaces");
105
106 // runtime checks
107 EXPECTS(mat_a.extent(1) == mat_b.extent(0));
108 EXPECTS(mat_a.extent(0) == c.extent(0));
109 EXPECTS(mat_b.extent(1) == c.extent(1));
110 EXPECTS(mat_a.indexmap().min_stride() == 1);
111 EXPECTS(mat_b.indexmap().min_stride() == 1);
112 EXPECTS(c.indexmap().min_stride() == 1);
113
114 // c is in C order: compute the transpose of the product in Fortran order
115 if constexpr (has_C_layout<C>) {
116 gemm(alpha, transpose(b), transpose(a), beta, transpose(std::forward<C>(c)));
117 } else { // c is in Fortran order
118 static constexpr bool conj_A = is_conj_array_expr<A>;
119 static constexpr bool conj_B = is_conj_array_expr<B>;
120 char op_a = get_op<conj_A, /* transpose = */ has_C_layout<mat_a_type>>;
121 char op_b = get_op<conj_B, /* transpose = */ has_C_layout<mat_b_type>>;
122 auto [m, k] = mat_a.shape();
123 auto n = mat_b.extent(1);
124
125 if constexpr (mem::have_device_compatible_addr_space<mat_a_type, mat_b_type, C>) {
126#if defined(NDA_HAVE_DEVICE)
127 device::gemm(op_a, op_b, m, n, k, alpha, mat_a.data(), get_ld(mat_a), mat_b.data(), get_ld(mat_b), beta, c.data(), get_ld(c));
128#else
129 compile_error_no_gpu();
130#endif
131 } else {
132 f77::gemm(op_a, op_b, m, n, k, alpha, mat_a.data(), get_ld(mat_a), mat_b.data(), get_ld(mat_b), beta, c.data(), get_ld(c));
133 }
134 }
