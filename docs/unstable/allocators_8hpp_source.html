123 template <int ChunkSize>
124 class bucket {
125 // Unique pointer to handle memory allocation for the bucket.
126 std::unique_ptr<char[]> _start = std::make_unique<char[]>(TotalChunkSize); // NOLINT (C-style array is fine here)
127
128 // Pointer to the start of the bucket.
129 char *p = _start.get();
130
131 // Bitmask to keep track of which chunks are free.
132 uint64_t flags = uint64_t(-1);
133
134 public:
136 static constexpr int TotalChunkSize = 64 * ChunkSize;
137
139 static constexpr auto address_space = Host;
140
141#ifdef NDA_USE_ASAN
142 bucket() { __asan_poison_memory_region(p, TotalChunkSize); }
143 ~bucket() { __asan_unpoison_memory_region(p, TotalChunkSize); }
144#else
146 bucket() = default;
147#endif
149 bucket(bucket const &) = delete;
150
152 bucket(bucket &&) = default;
153
155 bucket &operator=(bucket const &) = delete;
156
158 bucket &operator=(bucket &&) = default;
159
166 blk_t allocate(size_t s) noexcept {
167 // check the size and if there is a free chunk, otherwise abort
168 if (s > ChunkSize) std::abort();
169 if (flags == 0) std::abort();
170
171 // find the first free chunk
172 int pos = __builtin_ctzll(flags);
173
174 // update the bitmask and return the memory block
175 flags &= ~(1ull << pos);
176 blk_t b{p + static_cast<ptrdiff_t>(pos * ChunkSize), s};
177#ifdef NDA_USE_ASAN
178 __asan_unpoison_memory_region(b.ptr, ChunkSize);
179#endif
180 return b;
181 }
182
189 blk_t allocate_zero(size_t s) noexcept {
190 auto blk = allocate(s);
191 std::memset(blk.ptr, 0, s);
192 return blk;
193 }
194
199 void deallocate(blk_t b) noexcept {
200#ifdef NDA_USE_ASAN
201 __asan_poison_memory_region(b.ptr, ChunkSize);
202#endif
203 int pos = (b.ptr - p) / ChunkSize;
204 flags |= (1ull << pos);
205 }
206
211 [[nodiscard]] bool is_full() const noexcept { return flags == 0; }
212
217 [[nodiscard]] bool empty() const noexcept { return flags == uint64_t(-1); }
218
223 [[nodiscard]] const char *data() const noexcept { return p; }
224
229 [[nodiscard]] auto mask() const noexcept { return flags; }
230
237 [[nodiscard]] bool owns(blk_t b) const noexcept { return b.ptr >= p and b.ptr < p + TotalChunkSize; }
238 };

250 template <int ChunkSize>
251 class multi_bucket {
252 // Alias for the bucket allocator type.
253 using b_t = bucket<ChunkSize>;
254
255 // Vector of nda::mem::bucket allocators (ordered in memory).
256 std::vector<b_t> bu_vec;
257
258 // Iterator to the current bucket in use.
259 typename std::vector<b_t>::iterator bu;
260
261 // Find a bucket with a free chunk, otherwise create a new one and insert it at the correct position.
262 [[gnu::noinline]] void find_non_full_bucket() {
263 bu = std::find_if(bu_vec.begin(), bu_vec.end(), [](auto const &b) { return !b.is_full(); });
264 if (bu != bu_vec.end()) return;
265 b_t b;
266 auto pos = std::upper_bound(bu_vec.begin(), bu_vec.end(), b, [](auto const &b1, auto const &b2) { return b1.data() < b2.data(); });
267 bu = bu_vec.insert(pos, std::move(b));
268 }
269
270 public:
272 static constexpr auto address_space = Host;
273
275 multi_bucket() : bu_vec(1), bu(bu_vec.begin()) {}
276
278 multi_bucket(multi_bucket const &) = delete;
279
281 multi_bucket(multi_bucket &&) = default;
282
284 multi_bucket &operator=(multi_bucket const &) = delete;
285
287 multi_bucket &operator=(multi_bucket &&) = default;
288
295 blk_t allocate(size_t s) noexcept {
296 if ((bu == bu_vec.end()) or (bu->is_full())) find_non_full_bucket();
297 return bu->allocate(s);
298 }
299
307 blk_t allocate_zero(size_t s) noexcept {
308 auto blk = allocate(s);
309 std::memset(blk.ptr, 0, s);
310 return blk;
311 }
312
321 void deallocate(blk_t b) noexcept {
322 // try the current bucket first
323 if (bu != bu_vec.end() and bu->owns(b)) {
324 bu->deallocate(b);
325 return;
326 }
327
328 // otherwise, find the owning bucket in the vector and deallocate
329 bu = std::lower_bound(bu_vec.begin(), bu_vec.end(), b.ptr, [](auto const &b1, auto p) { return b1.data() <= p; });
330 --bu;
331 EXPECTS_WITH_MESSAGE((bu != bu_vec.end()), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
332 EXPECTS_WITH_MESSAGE((bu->owns(b)), "Error in nda::mem::multi_bucket::deallocate: Owning bucket not found");
333 bu->deallocate(b);
334
335 // remove bucket the current bucket if it is empty and not the only one
336 if (!bu->empty()) return;
337 if (bu_vec.size() <= 1) return;
338 bu_vec.erase(bu);
339 bu = bu_vec.end();
340 }
341
346 [[nodiscard]] bool empty() const noexcept { return bu_vec.size() == 1 && bu_vec[0].empty(); }
347
352 [[nodiscard]] auto const &buckets() const noexcept { return bu_vec; }
353
360 [[nodiscard]] bool owns(blk_t b) const noexcept {
361 bool res = false;
362 for (const auto &mb : bu_vec) { res = res || mb.owns(b); }
363 return res;
364 }
365 };
