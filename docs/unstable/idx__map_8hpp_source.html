188 [[nodiscard]] bool is_contiguous() const noexcept {
189 auto s = size();
190 if (s == 0) return true;
191 return (std::abs(str[stride_order[0]] * len[stride_order[0]]) == s);

198 [[nodiscard]] bool has_positive_strides() const noexcept { return (*std::min_element(str.cbegin(), str.cend()) >= 0); }

210 [[nodiscard]] bool is_strided_1d() const noexcept {
211 auto s = size();
212 if (s == 0) return true;
213 int i = Rank - 1;
214 while (len[stride_order[i]] == 1 and i > 0) --i;
215 return (std::abs(str[stride_order[0]] * len[stride_order[0]]) == s * std::abs(str[stride_order[i]]));
216 }

238 template <std::integral Int>
239 [[nodiscard]] static bool is_stride_order_valid(Int *lenptr, Int *strptr) {
240 auto dims_to_check = std::vector<int>{};
241 dims_to_check.reserve(Rank);
242 for (auto dim : stride_order)
243 if (lenptr[dim] > 1) dims_to_check.push_back(dim);
244 for (int n = 1; n < dims_to_check.size(); ++n)
245 if (std::abs(strptr[dims_to_check[n - 1]]) < std::abs(strptr[dims_to_check[n]])) return false;
246 return true;
247 }

254 [[nodiscard]] bool is_stride_order_valid() const { return is_stride_order_valid(len.data(), str.data()); }

257 // Compute contiguous strides from the shape.
258 void compute_strides_contiguous() {
259 long s = 1;
260 for (int v = rank() - 1; v >= 0; --v) {
261 int u = stride_order[v];
262 str[u] = s;
263 s *= len[u];
264 }
265 ENSURES(s == size());
266 }

441 // Get the contribution to the linear index in case of an nda::ellipsis argument.
442 template <bool skip_stride, auto I>
443 [[nodiscard]] FORCEINLINE long myget(ellipsis) const noexcept {
444 // nda::ellipsis are skipped and do not contribute to the linear index
445 return 0;
446 }
447
448 // Get the contribution to the linear index in case of a long argument.
449 template <bool skip_stride, auto I>
450 [[nodiscard]] FORCEINLINE long myget(long arg) const noexcept {
451 if constexpr (skip_stride and (I == stride_order[Rank - 1])) {
452 // optimize for the case when the fastest varying dimension is contiguous in memory
453 return arg;
454 } else {
455 // otherwise multiply the argument by the stride of the current dimension
456 return arg * std::get<I>(str);
457 }
458 }
459
460 // Is the smallest stride equal to one, i.e. is the fastest varying dimension contiguous in memory?
461 static constexpr bool smallest_stride_is_one = has_smallest_stride_is_one(layout_prop);
462
463 // Implementation of the function call operator that takes a multi-dimensional index and maps it to a linear index.
464 template <typename... Args, size_t... Is>
465 [[nodiscard]] FORCEINLINE long call_impl(std::index_sequence<Is...>, Args... args) const noexcept {
466 // set e_pos to the position of the ellipsis, otherwise to -1
467 static constexpr int e_pos = ((std::is_same_v<Args, ellipsis> ? int(Is) + 1 : 0) + ...) - 1;
468
469 if constexpr (e_pos == -1) {
470 // no ellipsis present
471 if constexpr (smallest_stride_is_one) {
472 // optimize for the case that the fastest varying dimension is contiguous in memory
473 return (myget<true, Is>(static_cast<long>(args)) + ...);
474 } else {
475 // arbitrary layouts
476 return ((args * std::get<Is>(str)) + ...);
477 }
478 } else {
479 // empty ellipsis is present and needs to be skipped
480 return (myget<smallest_stride_is_one, (Is < e_pos ? Is : Is - 1)>(args) + ...);
481 }
482 }

535 std::array<long, Rank> to_idx(long lin_idx) const {
536 // compute residues starting from slowest index
537 std::array<long, Rank> residues;
538 residues[0] = lin_idx;
539 for (auto i : range(1, Rank)) { residues[i] = residues[i - 1] % str[stride_order[i - 1]]; }
540
541 // convert residues to indices, ordered from slowest to fastest
542 std::array<long, Rank> idx;
543 idx[Rank - 1] = residues[Rank - 1] / str[stride_order[Rank - 1]];
544 for (auto i : range(Rank - 2, -1, -1)) { idx[i] = (residues[i] - residues[i + 1]) / str[stride_order[i]]; }
545
546 // reorder indices according to stride order
547 return permutations::apply_inverse(stride_order, idx);
548 }

595 template <uint64_t Permutation>
596 auto transpose() const {
597 // Makes a new transposed idx_map with permutation P such that
598 // denoting here A = this, A' = P A = returned_value
599 // A'(i_k) = A(i_{P[k]})
600 //
601 // Note that this convention is the correct one to have a (left) action of the symmetric group on
602 // a array and it may not be completely obvious.
603 // Proof
604 // let's operate with P then Q, and denote A'' = Q A'. We want to show that A'' = (QP) A
605 // A'(i_k) = A(i_{P[k]})
606 // A''(j_k) = A'(j_{Q[k]})
607 // then i_k = j_{Q[k]} and A''(j_k) = A(i_{P[k]}) = A(j_{Q[P[k]]}) = A(j_{(QP)[k]}), q.e.d
608 //
609 // NB test will test this composition
610 // Denoting this as A, an indexmap, calling it returns the linear index given by
611 //
612 // A(i_k) = sum_k i_k * S[k] (1)
613 //
614 // where S[k] denotes the strides.
615 //
616 // 1- S' : strides of A'
617 // A'(i_k) = sum_k i_{P[k]} * S[k] = sum_k i_k * S[P{^-1}[k]]
618 // so
619 // S'[k] = S[P{^-1}[k]] (2)
620 // i.e. apply (inverse(P), S) or apply_inverse directly.
621 //
622 // 2- L' : lengths of A'
623 // if L[k] is the k-th length, then because of the definition of A', i.e. A'(i_k) = A(i_{P[k]})
624 // i_q in the lhs A is at position q' such that P[q'] = q (A'(i0 i1 i2...) = A( i_P0 i_P1 i_P2....)
625 // hence L'[q] = L[q'] = L[P^{-1}[q]]
626 // same for static length
627 //
628 // 3- stride_order: denoted in this paragraph as Q (and Q' for A').
629 // by definition Q is a permutation such that Q[0] is the slowest index, Q[Rank -1] the fastest
630 // hence S[Q[k]] is a strictly decreasing sequence (as checked by strides_compatible_to_stride_order)
631 // we want therefore Q' the permutation that will sort the S', i.e.
632 // S'[Q'[k]] = S[Q[k]]
633 // using (2), we have S[P{^-1}[Q'[k]]] = S[Q[k]]
634 // so the permutation Q' is such that P{^-1}Q' = Q or Q' = PQ (as permutation product/composition).
635 // NB : Q and P are permutations, so the operation must be a composition, not an apply (apply applies
636 // a P to any set, like L, S, not only a permutation) even though they are all std::array in the code ...
637 //
638 static constexpr std::array<int, Rank> permu = decode<Rank>(Permutation);
639 static constexpr std::array<int, Rank> new_stride_order = permutations::compose(permu, stride_order);
640 static constexpr std::array<int, Rank> new_static_extents = permutations::apply_inverse(permu, static_extents);
641
642 return idx_map<Rank, encode(new_static_extents), encode(new_stride_order), LayoutProp>{permutations::apply_inverse(permu, lengths()),
643 permutations::apply_inverse(permu, strides())};
644 }
