35 /// Default accessor for various array and view types.
36 struct default_accessor {
37 /**
38 * @brief Accessor type of the nda::default_accessor.
39 * @tparam T Value type of the data.
40 */
41 template <typename T>
42 struct accessor {
43 /// Value type of the data.
44 using element_type = T;
45
46 /// Pointer type to the data.
47 using pointer = T *;
48
49 /// Reference type to the data.
50 using reference = T &;
51
52 /**
53 * @brief Access a specific element of the data.
54 *
55 * @param p Pointer to the data.
56 * @param i Index of the element to access.
57 * @return Reference to the element.
58 */
59 FORCEINLINE static reference access(pointer p, std::ptrdiff_t i) noexcept {
60 EXPECTS(p != nullptr);
61 return p[i];
62 }
63
64 /**
65 * @brief Offset the pointer by a certain number of elements.
66 *
67 * @param p Pointer to the data.
68 * @param i Number of elements to offset the pointer by.
69 * @return Pointer after applying the offset.
70 */
71 FORCEINLINE static T *offset(pointer p, std::ptrdiff_t i) noexcept { return p + i; }
72 };
73 };
74
75 /// Accessor for array and view types with no aliasing.
76 struct no_alias_accessor {
77 /**
78 * @brief Accessor type of the nda::no_alias_accessor.
79 * @tparam T Value type of the data.
80 */
81 template <typename T>
82 struct accessor {
83 /// Value type of the data.
84 using element_type = T;
85
86 /// Restricted pointer type to the data.
87 using pointer = T *__restrict;
88
89 /// Reference type to the data.
90 using reference = T &;
91
92 /**
93 * @brief Access a specific element of the data.
94 *
95 * @param p Pointer to the data.
96 * @param i Index of the element to access.
97 * @return Reference to the element.
98 */
99 FORCEINLINE static reference access(pointer p, std::ptrdiff_t i) noexcept { return p[i]; }
100
101 /**
102 * @brief Offset the pointer by a certain number of elements.
103 *
104 * @param p Pointer to the data.
105 * @param i Number of elements to offset the pointer by.
106 * @return Pointer after applying the offset.
107 */
108 FORCEINLINE static T *offset(pointer p, std::ptrdiff_t i) noexcept { return p + i; }
109 };
110 };
