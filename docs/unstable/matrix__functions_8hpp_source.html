59 template <ArrayOfRank<2> M>
60 auto trace(M const &m) {
61 static_assert(get_rank<M> == 2, "Error in nda::trace: Array/View must have rank 2");
62 EXPECTS(m.shape()[0] == m.shape()[1]);
63 auto r = get_value_t<M>{};
64 auto d = m.shape()[0];
65 for (int i = 0; i < d; ++i) r += m(i, i);
66 return r;
67 }

114 if constexpr (std::ranges::contiguous_range<V>) {
115 return diag(nda::basic_array_view{v});

151 template <ArrayOfRank<2> A, ArrayOfRank<2> B>
152 requires(std::same_as<get_value_t<A>, get_value_t<B>>) // NB the get_value_t gets rid of const if any
153 matrix<get_value_t<A>> vstack(A const &a, B const &b) {
154 static_assert(get_rank<A> == 2, "Error in nda::vstack: Only rank 2 arrays/views are allowed");
155 static_assert(get_rank<A> == 2, "Error in nda::vstack: Only rank 2 arrays/views are allowed");
156 EXPECTS_WITH_MESSAGE(a.shape()[1] == b.shape()[1], "Error in nda::vstack: The second dimension of the two matrices must be equal");
157
158 auto [n, q] = a.shape();
159 auto p = b.shape()[0];
160 matrix<get_value_t<A>> res(n + p, q);
161 res(range(0, n), range::all) = a;
162 res(range(n, n + p), range::all) = b;
163 return res;
164 }

175 template <Matrix A>
176 bool is_matrix_square(A const &a, bool print_error = false) {
177 auto const [m, n] = a.shape();
178 if (m != n and print_error) std::cerr << "Error in nda::is_matrix_square: Dimensions are: (" << m << "," << n << ")\n" << std::endl;
179 return m == n;
180 }

191 template <Matrix A>
192 bool is_matrix_diagonal(A const &a, bool print_error = false) {
193 bool const r = is_matrix_square(a) and a == diag(diagonal(a));
194 if (not r and print_error) std::cerr << "Error in nda::is_matrix_diagonal: Non-diagonal matrix: " << a << std::endl;
195 return r;
196 }
