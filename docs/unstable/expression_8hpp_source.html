65 /**
66 * @brief Single node of the expression tree.
67 *
68 * @details An expression node contains a tag that determines the type of expression and a tuple of child nodes which
69 * are usually either other expression nodes, nda::clef::placeholder objects or specific values/objects like an int or
70 * a double.
71 *
72 * @tparam Tag Tag of the expression.
73 * @tparam Ts Types of the child nodes.
74 */
75 template <typename Tag, typename... Ts>
76 struct expr {
77 /// Tuple type for storing the child nodes.
78 using childs_t = std::tuple<Ts...>;
79
80 /// Child nodes of the current expression node.
81 childs_t childs;
82
83 /// Default copy constructor.
84 expr(expr const &) = default;
85
86 /**
87 * @brief Move constructor simply moves the child nodes from the source expression.
88 * @param ex Source expression.
89 */
90 expr(expr &&ex) noexcept : childs(std::move(ex.childs)) {}
91
92 /// Copy assignment operator is deleted.
93 expr &operator=(expr const &) = delete;
94
95 /// Default move assignment operator.
96 expr &operator=(expr &&) = default;
97
98 /**
99 * @brief Construct an expression node with a given tag and child nodes.
100 *
101 * @tparam Us Types of the child nodes.
102 * @param us Child nodes.
103 */
104 template <typename... Us>
105 expr(Tag, Us &&...us) : childs(std::forward<Us>(us)...) {}
106
107 /**
108 * @brief Subscript operator.
109 *
110 * @tparam Args Types of the subscript arguments.
111 * @param args Subscript arguments.
112 * @return An nda::clef::expr object with the nda::clef::tags::subscript tag containing the current expression node
113 * as the first child node and the other arguments as the remaining child nodes.
114 */
115 template <typename... Args>
116 auto operator[](Args &&...args) const {
117 return expr<tags::subscript, expr, expr_storage_t<Args>...>{tags::subscript(), *this, std::forward<Args>(args)...};
118 }
119
120 /**
121 * @brief Function call operator.
122 *
123 * @tparam Args Types of the function call arguments.
124 * @param args Function call arguments.
125 * @return An nda::clef::expr object with the nda::clef::tags::function tag containing the current expression node
126 * as the first child node and the other arguments as the remaining child nodes.
127 */
128 template <typename... Args>
129 auto operator()(Args &&...args) const {
130 return expr<tags::function, expr, expr_storage_t<Args>...>{tags::function(), *this, std::forward<Args>(args)...};
131 }
132 };
