58 // Check if a scalar or either component of a complex scalar is NaN.
59 template <Scalar S>
60 bool isnan(S const &x) {
61 if constexpr (is_complex_v<S>) {
62 return std::isnan(x.real()) or std::isnan(x.imag());
63 } else {
64 return std::isnan(x);
65 }

127 template <ArrayOrScalar A>
128 auto reciprocal(A &&a) {
129 return nda::map([]<typename T>(T const &x) {
130 if constexpr (Scalar<T>) {
131 using real_t = remove_complex_t<T>;
132 if constexpr (std::is_floating_point_v<real_t>) {
133 return real_t{1} / x;
134 } else {
135 return 1.0 / x;
136 }
137 } else {
138 return nda::reciprocal(x);
139 }
140 })(std::forward<A>(a));
141 }
