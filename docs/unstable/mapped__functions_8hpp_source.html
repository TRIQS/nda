57 // Check if a scalar or either component of a complex scalar is NaN.
58 template <Scalar S>
59 bool isnan(S const &x) {
60 if constexpr (is_complex_v<S>) {
61 return std::isnan(x.real()) or std::isnan(x.imag());
62 } else {
63 return std::isnan(x);
64 }
65 }

116 template <ArrayOrScalar A>
117 auto reciprocal(A &&a) {
118 return nda::map([]<typename T>(T const &x) {
119 if constexpr (Scalar<T>) {
120 using real_t = remove_complex_t<T>;
121 if constexpr (std::is_floating_point_v<real_t>) {
122 return real_t{1} / x;
123 } else {
124 return 1.0 / x;
125 }
126 } else {
127 return nda::reciprocal(x);
128 }
129 })(std::forward<A>(a));
130 }
