43 // Helper function to get the shape and total size of the scattered array/view as well as the stride along the first
44 // dimension.
45 template <typename A>
46 requires(is_regular_or_view_v<A> and std::decay_t<A>::is_stride_order_C())
47 auto mpi_scatter_shape_impl(A const &a, mpi::communicator comm, int root) {
48 auto dims = a.shape();
49 mpi::broadcast(dims, comm, root);
50 auto scattered_size = std::accumulate(dims.begin(), dims.end(), 1, std::multiplies<>());
51 auto stride0 = scattered_size / dims[0];
52 dims[0] = mpi::chunk_length(dims[0], comm.size(), comm.rank());
53 return std::make_tuple(dims, scattered_size, stride0);
54 }

93 template <typename A1, typename A2>
94 requires(is_regular_or_view_v<A1> and std::decay_t<A1>::is_stride_order_C()
95 and is_regular_or_view_v<A2> and std::decay_t<A2>::is_stride_order_C())
96 void mpi_scatter_into(A1 const &a_in, A2 &&a_out, mpi::communicator comm = {}, int root = 0) { // NOLINT
97 // check the ranks of the input arrays/views
98 EXPECTS_WITH_MESSAGE(detail::have_mpi_equal_ranks(a_in, comm), "Error in nda::mpi_scatter_into: Ranks of arrays/views must be equal")
99
100 // check if the input and output arrays/views can be used in the MPI call
101 if (comm.rank() == root) detail::check_layout_mpi_compatible(a_in, "mpi_scatter_into");
102 detail::check_layout_mpi_compatible(a_out, "mpi_scatter_into");
103
104 // get output shape and resize or check the output array/view
105 auto [dims, scattered_size, stride0] = detail::mpi_scatter_shape_impl(a_in, comm, root);
106 resize_or_check_if_view(a_out, dims);
107
108 // scatter the data
109 auto a_out_span = std::span{a_out.data(), static_cast<std::size_t>(a_out.size())};
110 auto a_in_span = std::span{a_in.data(), static_cast<std::size_t>(a_in.size())};
111 mpi::scatter_range(a_in_span, a_out_span, scattered_size, comm, root, stride0);
112 }

130 template <typename A>
131 requires(is_regular_or_view_v<A> and std::decay_t<A>::is_stride_order_C())
132 auto mpi_scatter(A const &a, mpi::communicator comm = {}, int root = 0) {
133 using return_t = get_regular_t<A>;
134 return_t a_out;
135 mpi::scatter_into(a, a_out, comm, root);
