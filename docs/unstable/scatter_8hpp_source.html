43 // Helper function to get the shape and total size of the scattered array/view.
44 template <typename A>
45 requires(is_regular_or_view_v<A> and std::decay_t<A>::is_stride_order_C())
46 auto mpi_scatter_shape_impl(A const &a, mpi::communicator comm, int root) {
47 auto dims = a.shape();
48 mpi::broadcast(dims, comm, root);
49 auto scattered_size = std::accumulate(dims.begin(), dims.end(), 1, std::multiplies<>());
50 auto stride0 = scattered_size / dims[0];
51 dims[0] = mpi::chunk_length(dims[0], comm.size(), comm.rank());
52 return std::make_tuple(dims, scattered_size, stride0);
53 }

55 // Helper function that scatters arrays/views.
56 template <typename A1, typename A2>
57 requires(is_regular_or_view_v<A1> and std::decay_t<A1>::is_stride_order_C()
58 and is_regular_or_view_v<A2> and std::decay_t<A2>::is_stride_order_C())
59 void mpi_scatter_impl(A1 const &a_in, A2 &&a_out, mpi::communicator comm = {}, int root = 0) { // NOLINT
60 // check the ranks of the input arrays/views
61 EXPECTS_WITH_MESSAGE(detail::have_mpi_equal_ranks(a_in, comm), "Error in nda::detail::mpi_scatter_impl: Ranks of arrays/views must be equal")
62
63 // simply copy if there is no active MPI environment or if the communicator size is < 2
64 if (not mpi::has_env) {
65 a_out = a_in;
66 return;
67 }
68
69 // check if the input and output arrays/views can be used in the MPI call
70 if (comm.rank() == root) detail::check_layout_mpi_compatible(a_in, "detail::mpi_scatter_impl");
71 check_layout_mpi_compatible(a_out, "detail::mpi_scatter_impl");
72
73 // get output shape and resize or check the output array/view
74 auto [dims, scattered_size, stride0] = mpi_scatter_shape_impl(a_in, comm, root);
75 resize_or_check_if_view(a_out, dims);
76
77 // scatter the data
78 auto a_out_span = std::span{a_out.data(), static_cast<std::size_t>(a_out.size())};
79 auto a_in_span = std::span{a_in.data(), static_cast<std::size_t>(a_in.size())};
80 mpi::scatter_range(a_in_span, a_out_span, scattered_size, comm, root, stride0);
81 }
