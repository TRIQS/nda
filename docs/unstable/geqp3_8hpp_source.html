81 int geqp3(A &&a, JPVT &&jpvt, TAU &&tau, W1 &&work = vector_value_t<A>{}, W2 &&rwork = vector_fp_t<A>{}) { // NOLINT
82 // check the dimensions of the input/output arrays/views and resize if necessary
83 auto const [m, n] = a.shape();
84 EXPECTS(jpvt.size() == n);
85 resize_or_check_if_view(tau, {std::min(m, n)});
86 resize_or_check_work_buffer(rwork, 2 * n);
87
88 // arrays/views must be LAPACK compatible
89 EXPECTS(a.indexmap().min_stride() == 1);
90 EXPECTS(jpvt.indexmap().min_stride() == 1);
91 EXPECTS(tau.indexmap().min_stride() == 1);
92
93 // first call to get the optimal buffer size
94 auto tmp_lwork = get_value_t<A>{};
95 int info = 0;
96 lapack::f77::geqp3(m, n, a.data(), get_ld(a), jpvt.data(), tau.data(), &tmp_lwork, -1, rwork.data(), info);
97 int lwork = static_cast<int>(std::ceil(std::real(tmp_lwork)));
98
99 // resize/check work buffer
100 resize_or_check_work_buffer(work, lwork);
101
102 // perform actual library call
103 lapack::f77::geqp3(m, n, a.data(), get_ld(a), jpvt.data(), tau.data(), work.data(), lwork, rwork.data(), info);
104
105 return info;
106 }
