53 template <MemoryVector X, MemoryVector Y, MemoryMatrix M>
54 requires(have_same_value_type_v<X, Y, M> and mem::have_compatible_addr_space<X, Y, M> and is_blas_lapack_v<get_value_t<X>>)
55 void ger(get_value_t<X> alpha, X const &x, Y const &y, M &&m) { // NOLINT (temporary views are allowed here)
56 // for C-layout arrays/views, call ger with the transpose and swap x and y
57 if constexpr (has_C_layout<M>) {
58 ger(alpha, y, x, transpose(m));
59 return;
60 }
61
62 // check the dimensions of the input/output arrays/views
63 EXPECTS(m.extent(0) == x.size());
64 EXPECTS(m.extent(1) == y.size());
65
66 // arrays/views must be BLAS compatible
67 EXPECTS(m.indexmap().min_stride() == 1);
68
69 // perform actual library call
70 if constexpr (mem::have_device_compatible_addr_space<X, Y, M>) {
71#if defined(NDA_HAVE_DEVICE)
72 device::ger(m.extent(0), m.extent(1), alpha, x.data(), x.indexmap().strides()[0], y.data(), y.indexmap().strides()[0], m.data(), get_ld(m));
73#else
74 compile_error_no_gpu();
75#endif
76 } else {
77 f77::ger(m.extent(0), m.extent(1), alpha, x.data(), x.indexmap().strides()[0], y.data(), y.indexmap().strides()[0], m.data(), get_ld(m));
78 }
79 }
