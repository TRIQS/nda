35 template <typename Allocator =
36 mem::segregator<8 * NDA_TEST_DEFAULT_ALLOC_MBUCKET, mem::multi_bucket<8 * NDA_TEST_DEFAULT_ALLOC_MBUCKET>, mem::mallocator>>
37#else // normal case
38 /**
39 * @brief Memory policy using an nda::mem::handle_heap.
40 * @tparam Allocator Allocator type to be used.
41 */
42 template <typename Allocator>
43#endif
44 struct heap_basic {
45 /**
46 * @brief Handle type for the policy.
47 * @tparam T Value type of the data.
48 */
49 template <typename T>
50#ifdef NDA_TEST_DEFAULT_ALLOC_SSO // for testing only: use a handle_sso if possible
51 using handle =
52 std::conditional_t<std::is_copy_constructible_v<T>, mem::handle_sso<T, NDA_TEST_DEFAULT_ALLOC_SSO>, mem::handle_heap<T, Allocator>>;
53#else
54 using handle = mem::handle_heap<T, Allocator>;
55#endif
56 };
57
58 /**
59 * @brief Alias template of the nda::heap_basic policy using an nda::mem::mallocator.
60 * @tparam AdrSp nda::mem::AddressSpace in which the memory is allocated.
61 */
62 template <mem::AddressSpace AdrSp = mem::Host>
63 using heap = heap_basic<mem::mallocator<AdrSp>>;

65 /**
66 * @brief Memory policy using an nda::mem::handle_sso.
67 * @tparam Size Max. size of the data to store on the stack (number of elements).

69 template <size_t Size>
70 struct sso {
71 /**
72 * @brief Handle type for the policy.
73 * @tparam T Value type of the data.
74 */
75 template <typename T>
76 using handle = mem::handle_sso<T, Size>;
77 };
78
79 /**
80 * @brief Memory policy using an nda::mem::handle_stack.

83 template <size_t Size>
84 struct stack {
85 /**
86 * @brief Handle type for the policy.
87 * @tparam T Value type of the data.
88 */
89 template <typename T>

91 };

98 */
