569template <char ResultAlgebra, bool SelfIsRvalue, typename Self, typename... Ts>
570FORCEINLINE static decltype(auto) call(Self &&self, Ts const &...idxs) noexcept(has_no_boundcheck) {
571 // resulting value type
572 using r_v_t = std::conditional_t<std::is_const_v<std::remove_reference_t<Self>>, ValueType const, ValueType>;
573
574 // behavior depends on the given arguments
575 if constexpr (clef::is_any_lazy<Ts...>) {
576 // if there are lazy arguments, e.g. as in A(i_) << i_, a lazy expression is returned
577 return clef::make_expr_call(std::forward<Self>(self), idxs...);
578 } else if constexpr (sizeof...(Ts) == 0) {
579 // if no arguments are given, a full view is returned
580 return basic_array_view<r_v_t, Rank, LayoutPolicy, Algebra, AccessorPolicy, OwningPolicy>{self.lay, self.sto};
581 } else {
582 // otherwise we check the arguments and either access a single element or make a slice
583 static_assert(((layout_t::template argument_is_allowed_for_call_or_slice<Ts> + ...) > 0),
584 "Error in array/view: Slice arguments must be convertible to range, ellipsis, or long (or string if the layout permits it)");
585
586 // number of arguments convertible to long
587 static constexpr int n_args_long = (layout_t::template argument_is_allowed_for_call<Ts> + ...);
588
589 if constexpr (n_args_long == rank) {
590 // access a single element
591 long offset = self.lay(idxs...);
592 if constexpr (is_view or not SelfIsRvalue) {
593 // if the calling object is a view or an lvalue, we return a reference
594 return AccessorPolicy::template accessor<r_v_t>::access(self.sto.data(), offset);
595 } else {
596 // otherwise, we return a copy of the value
597 return ValueType{self.sto[offset]};
598 }
599 } else {
600 // access a slice of the view/array
601 auto const [offset, idxm] = self.lay.slice(idxs...);
602 static constexpr auto res_rank = decltype(idxm)::rank();
603 // resulting algebra
604 static constexpr char newAlgebra = (ResultAlgebra == 'M' and (res_rank == 1) ? 'V' : ResultAlgebra);
605 // resulting layout policy
606 using r_layout_p = typename detail::layout_to_policy<std::decay_t<decltype(idxm)>>::type;
607 return basic_array_view<r_v_t, res_rank, r_layout_p, newAlgebra, AccessorPolicy, OwningPolicy>{std::move(idxm), {self.sto, offset}};
608 }
609 }
610}

704// Make an iterator for the view/array depending on its type.
705template <typename Iterator>
706[[nodiscard]] auto make_iterator(bool at_end) const noexcept {
707 if constexpr (iterator_rank == Rank) {
708 // multi-dimensional iterator
709 if constexpr (layout_t::is_stride_order_C()) {
710 // C-order case (array_iterator already traverses the data in C-order)
711 return Iterator{indexmap().lengths(), indexmap().strides(), sto.data(), at_end};
712 } else {
713 // general case (we need to permute the shape and the strides according to the stride order of the layout)
714 return Iterator{nda::permutations::apply(layout_t::stride_order, indexmap().lengths()),
715 nda::permutations::apply(layout_t::stride_order, indexmap().strides()), sto.data(), at_end};
716 }
717 } else {
718 // 1-dimensional iterator
719 return Iterator{std::array<long, 1>{size()}, std::array<long, 1>{indexmap().min_stride()}, sto.data(), at_end};
720 }
721}

831// Implementation of the assignment from an n-dimensional array type.
832template <typename RHS>
833void assign_from_ndarray(RHS const &rhs) {
834#ifdef NDA_ENFORCE_BOUNDCHECK
835 if (this->shape() != rhs.shape()) {
836 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Size mismatch:"
837 << "\n LHS.shape() = " << this->shape() << "\n RHS.shape() = " << rhs.shape();
838 }
839#endif
840 // compile-time check if assignment is possible
841 static_assert(std::is_assignable_v<value_type &, get_value_t<RHS>>, "Error in assign_from_ndarray: Incompatible value types");
842
843 // are both operands nda::MemoryArray types?
844 static constexpr bool both_in_memory = MemoryArray<self_t> and MemoryArray<RHS>;
845
846 // do both operands have the same stride order?
847 static constexpr bool same_stride_order = get_layout_info<self_t>.stride_order == get_layout_info<RHS>.stride_order;
848
849 // compile-time check for device arrays to avoid runtime errors
850 static_assert(!(mem::on_device<self_t> or mem::on_device<RHS>) or (both_in_memory and same_stride_order and have_same_value_type_v<self_t, RHS>),
851 "Error in assign_from_ndarray: Assignment to/from device arrays is not supported for the given types.");
852
853 // prefer optimized options if possible
854 if constexpr (both_in_memory and same_stride_order) {
855 if (rhs.empty()) return;
856 // are both operands strided in 1d?
857 static constexpr bool both_1d_strided = has_layout_strided_1d<self_t> and has_layout_strided_1d<RHS>;
858 if constexpr (mem::on_host<self_t, RHS> and both_1d_strided) {
859 // vectorizable copy on host
860 for (long i = 0; i < size(); ++i) (*this)(_linear_index_t{i}) = rhs(_linear_index_t{i});
861 return;
862 } else if constexpr (!mem::on_host<self_t, RHS> and have_same_value_type_v<self_t, RHS>) {
863 // check for block-layout and use mem::memcpy2D if possible
864 auto bl_layout_dst = get_block_layout(*this);
865 auto bl_layout_src = get_block_layout(rhs);
866 if (bl_layout_dst && bl_layout_src) {
867 auto [n_bl_dst, bl_size_dst, bl_str_dst] = *bl_layout_dst;
868 auto [n_bl_src, bl_size_src, bl_str_src] = *bl_layout_src;
869 // check that the total memory size is the same
870 if (n_bl_dst * bl_size_dst != n_bl_src * bl_size_src) NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Incompatible block sizes";
871 // if either destination or source consists of a single block, we can chunk it up to make the layouts compatible
872 if (n_bl_dst == 1 && n_bl_src > 1) {
873 n_bl_dst = n_bl_src;
874 bl_size_dst /= n_bl_src;
875 bl_str_dst = bl_size_dst;
876 }
877 if (n_bl_src == 1 && n_bl_dst > 1) {
878 n_bl_src = n_bl_dst;
879 bl_size_src /= n_bl_dst;
880 bl_str_src = bl_size_src;
881 }
882 // copy only if block-layouts are compatible, otherwise continue to fallback
883 if (n_bl_dst == n_bl_src && bl_size_dst == bl_size_src) {
884 mem::memcpy2D<mem::get_addr_space<self_t>, mem::get_addr_space<RHS>>((void *)data(), bl_str_dst * sizeof(value_type), (void *)rhs.data(),
885 bl_str_src * sizeof(value_type), bl_size_src * sizeof(value_type),
886 n_bl_src);
887 return;
888 }
889 }
890 }
891 }
892 // otherwise fallback to elementwise assignment
893 if constexpr (mem::on_device<self_t> || mem::on_device<RHS>) {
894 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Fallback to elementwise assignment not implemented for arrays/views on the GPU";
895 }
896 nda::for_each(shape(), [this, &rhs](auto const &...args) { (*this)(args...) = rhs(args...); });
897}

899// Implementation to fill a view/array with a constant scalar value.
900template <typename Scalar>
901void fill_with_scalar(Scalar const &scalar) {
902 if constexpr (mem::on_host<self_t>) {
903 if constexpr (has_layout_strided_1d<self_t>) {
904 const long L = size();
905 auto *__restrict const p = data(); // no alias possible here!
906 if constexpr (has_contiguous_layout<self_t>) {
907 for (long i = 0; i < L; ++i) p[i] = scalar;
908 } else {
909 const long stri = indexmap().min_stride();
910 const long Lstri = L * stri;
911 for (long i = 0; i != Lstri; i += stri) p[i] = scalar;
912 }
913 } else {
914 for (auto &x : *this) x = scalar;
915 }
916 } else if constexpr (mem::on_device<self_t> or mem::on_unified<self_t>) {
917 if constexpr (has_layout_strided_1d<self_t>) {
918 if constexpr (has_contiguous_layout<self_t>) {
919 mem::fill_n<mem::get_addr_space<self_t>>(data(), size(), value_type(scalar));
920 } else {
921 const long stri = indexmap().min_stride();
922 mem::fill2D_n<mem::get_addr_space<self_t>>(data(), stri, 1, size(), value_type(scalar));
923 }
924 } else {
925 auto bl_layout = get_block_layout(*this);
926 if (bl_layout) {
927 auto [n_bl, bl_size, bl_str] = *bl_layout;
928 mem::fill2D_n<mem::get_addr_space<self_t>>(data(), bl_str, bl_size, n_bl, value_type(scalar));
929 } else {
930 // MAM: implement recursive call to fill_with_scalar on (i,nda::ellipsis{})
931 NDA_RUNTIME_ERROR << "fill_with_scalar: Not implemented on device for generic (non-blocked) layout.";
932 }
933 }
934 }
935}

948 if constexpr (is_scalar_or_convertible_v<Scalar>)
949 fill_with_scalar(0);
950 else
951 fill_with_scalar(Scalar{0 * scalar}); // FIXME : improve this

953 }
