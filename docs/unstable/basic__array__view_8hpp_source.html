578template <char ResultAlgebra, bool SelfIsRvalue, typename Self, typename... Ts>
579FORCEINLINE static decltype(auto) call(Self &&self, Ts const &...idxs) noexcept(has_no_boundcheck) {
580 // resulting value type
581 using r_v_t = std::conditional_t<std::is_const_v<std::remove_reference_t<Self>>, ValueType const, ValueType>;
582
583 // behavior depends on the given arguments
584 if constexpr (clef::is_any_lazy<Ts...>) {
585 // if there are lazy arguments, e.g. as in A(i_) << i_, a lazy expression is returned
586 return clef::make_expr_call(std::forward<Self>(self), idxs...);
587 } else if constexpr (sizeof...(Ts) == 0) {
588 // if no arguments are given, a full view is returned
589 return basic_array_view<r_v_t, Rank, LayoutPolicy, Algebra, AccessorPolicy, OwningPolicy>{self.lay, self.sto};
590 } else if constexpr (detail::has_index_container<Ts...>) {
591 // if any argument is an IndexContainer, return an expr_indexed for advanced indexing
592 auto view = self(detail::make_view_arg_for_indexed(idxs)...);
593 auto idx_tuple = detail::filter_and_convert(idxs...);
594 constexpr auto indexed_dims = detail::compute_indexed_dims<Ts...>();
595 return expr_indexed{std::move(view), std::move(idx_tuple), indexed_dims};
596 } else {
597 // otherwise we check the arguments and either access a single element or make a slice
598 static_assert(((layout_t::template argument_is_allowed_for_call_or_slice<Ts> + ...) > 0),
599 "Error in array/view: Slice arguments must be convertible to range, ellipsis, or long (or string if the layout permits it)");
600
601 // number of arguments convertible to long
602 static constexpr int n_args_long = (layout_t::template argument_is_allowed_for_call<Ts> + ...);
603
604 if constexpr (n_args_long == rank) {
605 // access a single element
606 long offset = self.lay(idxs...);
607 if constexpr (is_view or not SelfIsRvalue) {
608 // if the calling object is a view or an lvalue, we return a reference
609 return AccessorPolicy::template accessor<r_v_t>::access(self.sto.data(), offset);
610 } else {
611 // otherwise, we return a copy of the value
612 return ValueType{self.sto[offset]};
613 }
614 } else {
615 // access a slice of the view/array
616 auto const [offset, idxm] = self.lay.slice(idxs...);
617 static constexpr auto res_rank = decltype(idxm)::rank();
618 // resulting algebra
619 static constexpr char newAlgebra = (ResultAlgebra == 'M' and (res_rank == 1) ? 'V' : ResultAlgebra);
620 // resulting layout policy
621 using r_layout_p = typename detail::layout_to_policy<std::decay_t<decltype(idxm)>>::type;
622 return basic_array_view<r_v_t, res_rank, r_layout_p, newAlgebra, AccessorPolicy, OwningPolicy>{std::move(idxm), {self.sto, offset}};
623 }
624 }
625}

719// Make an iterator for the view/array depending on its type.
720template <typename Iterator>
721[[nodiscard]] auto make_iterator(bool at_end) const noexcept {
722 if constexpr (iterator_rank == Rank) {
723 // multi-dimensional iterator
724 if constexpr (layout_t::is_stride_order_C()) {
725 // C-order case (array_iterator already traverses the data in C-order)
726 return Iterator{indexmap().lengths(), indexmap().strides(), sto.data(), at_end};
727 } else {
728 // general case (we need to permute the shape and the strides according to the stride order of the layout)
729 return Iterator{nda::permutations::apply(layout_t::stride_order, indexmap().lengths()),
730 nda::permutations::apply(layout_t::stride_order, indexmap().strides()), sto.data(), at_end};
731 }
732 } else {
733 // 1-dimensional iterator
734 return Iterator{std::array<long, 1>{size()}, std::array<long, 1>{indexmap().min_stride()}, sto.data(), at_end};
735 }
736}

846// Implementation of the assignment from an n-dimensional array type.
847template <typename RHS>
848void assign_from_ndarray(RHS const &rhs) {
849#ifdef NDA_ENFORCE_BOUNDCHECK
850 if (this->shape() != rhs.shape()) {
851 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Size mismatch:"
852 << "\n LHS.shape() = " << this->shape() << "\n RHS.shape() = " << rhs.shape();
853 }
854#endif
855 // compile-time check if assignment is possible
856 static_assert(std::is_assignable_v<value_type &, get_value_t<RHS>>, "Error in assign_from_ndarray: Incompatible value types");
857
858 // are both operands nda::MemoryArray types?
859 static constexpr bool both_in_memory = MemoryArray<self_t> and MemoryArray<RHS>;
860
861 // do both operands have the same stride order?
862 static constexpr bool same_stride_order = get_layout_info<self_t>.stride_order == get_layout_info<RHS>.stride_order;
863
864 // compile-time check for device arrays to avoid runtime errors
865 static_assert(!(mem::on_device<self_t> or mem::on_device<RHS>) or (both_in_memory and same_stride_order and have_same_value_type_v<self_t, RHS>),
866 "Error in assign_from_ndarray: Assignment to/from device arrays is not supported for the given types.");
867
868 // prefer optimized options if possible
869 if constexpr (both_in_memory and same_stride_order) {
870 if (rhs.empty()) return;
871 // are both operands strided in 1d?
872 static constexpr bool both_1d_strided = has_layout_strided_1d<self_t> and has_layout_strided_1d<RHS>;
873 if constexpr (mem::on_host<self_t, RHS> and both_1d_strided) {
874 // vectorizable copy on host
875 for (long i = 0; i < size(); ++i) (*this)(_linear_index_t{i}) = rhs(_linear_index_t{i});
876 return;
877 } else if constexpr (!mem::on_host<self_t, RHS> and have_same_value_type_v<self_t, RHS>) {
878 // check for block-layout and use mem::memcpy2D if possible
879 auto bl_layout_dst = get_block_layout(*this);
880 auto bl_layout_src = get_block_layout(rhs);
881 if (bl_layout_dst && bl_layout_src) {
882 auto [n_bl_dst, bl_size_dst, bl_str_dst] = *bl_layout_dst;
883 auto [n_bl_src, bl_size_src, bl_str_src] = *bl_layout_src;
884 // check that the total memory size is the same
885 if (n_bl_dst * bl_size_dst != n_bl_src * bl_size_src) NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Incompatible block sizes";
886 // if either destination or source consists of a single block, we can chunk it up to make the layouts compatible
887 if (n_bl_dst == 1 && n_bl_src > 1) {
888 n_bl_dst = n_bl_src;
889 bl_size_dst /= n_bl_src;
890 bl_str_dst = bl_size_dst;
891 }
892 if (n_bl_src == 1 && n_bl_dst > 1) {
893 n_bl_src = n_bl_dst;
894 bl_size_src /= n_bl_dst;
895 bl_str_src = bl_size_src;
896 }
897 // copy only if block-layouts are compatible, otherwise continue to fallback
898 if (n_bl_dst == n_bl_src && bl_size_dst == bl_size_src) {
899 mem::memcpy2D<mem::get_addr_space<self_t>, mem::get_addr_space<RHS>>((void *)data(), bl_str_dst * sizeof(value_type), (void *)rhs.data(),
900 bl_str_src * sizeof(value_type), bl_size_src * sizeof(value_type),
901 n_bl_src);
902 return;
903 }
904 }
905 }
906 }
907 // otherwise fallback to elementwise assignment
908 if constexpr (mem::on_device<self_t> || mem::on_device<RHS>) {
909 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Fallback to elementwise assignment not implemented for arrays/views on the GPU";
910 }
911 nda::for_each(shape(), [this, &rhs](auto const &...args) { (*this)(args...) = rhs(args...); });
912}

914// Implementation to fill a view/array with a constant scalar value.
915template <typename Scalar>
916void fill_with_scalar(Scalar const &scalar) {
917 if constexpr (mem::on_host<self_t>) {
918 if constexpr (has_layout_strided_1d<self_t>) {
919 const long L = size();
920 auto *__restrict const p = data(); // no alias possible here!
921 if constexpr (has_contiguous_layout<self_t>) {
922 for (long i = 0; i < L; ++i) p[i] = scalar;
923 } else {
924 const long stri = indexmap().min_stride();
925 const long Lstri = L * stri;
926 for (long i = 0; i != Lstri; i += stri) p[i] = scalar;
927 }
928 } else {
929 for (auto &x : *this) x = scalar;
930 }
931 } else if constexpr (mem::on_device<self_t> or mem::on_unified<self_t>) {
932 if constexpr (has_layout_strided_1d<self_t>) {
933 if constexpr (has_contiguous_layout<self_t>) {
934 mem::fill_n<mem::get_addr_space<self_t>>(data(), size(), value_type(scalar));
935 } else {
936 const long stri = indexmap().min_stride();
937 mem::fill2D_n<mem::get_addr_space<self_t>>(data(), stri, 1, size(), value_type(scalar));
938 }
939 } else {
940 auto bl_layout = get_block_layout(*this);
941 if (bl_layout) {
942 auto [n_bl, bl_size, bl_str] = *bl_layout;
943 mem::fill2D_n<mem::get_addr_space<self_t>>(data(), bl_str, bl_size, n_bl, value_type(scalar));
944 } else {
945 // MAM: implement recursive call to fill_with_scalar on (i,nda::ellipsis{})
946 NDA_RUNTIME_ERROR << "fill_with_scalar: Not implemented on device for generic (non-blocked) layout.";
947 }
948 }
949 }
950}

954void assign_from_scalar(Scalar const &scalar) noexcept {
955 static_assert(!is_const, "Error in assign_from_ndarray: Cannot assign to a const view");
956 if constexpr (Algebra != 'M') {
957 // element-wise assignment for non-matrix algebras
958 fill_with_scalar(scalar);
959 } else {
960 // a scalar has to be interpreted as a unit matrix for matrix algebras (the scalar in the shortest diagonal)
961 // FIXME : A priori faster to put 0 everywhere and then change the diag to avoid the if.
962 // FIXME : Benchmark and confirm.
963 if constexpr (is_scalar_or_convertible_v<Scalar>)
964 fill_with_scalar(0);
965 else
966 fill_with_scalar(Scalar{0 * scalar}); // FIXME : improve this
967 diagonal(*this).fill_with_scalar(scalar);
968 }
969}
