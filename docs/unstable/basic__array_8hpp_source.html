272 template <ArrayOfRank<Rank> A>
273 requires(HasValueTypeConstructibleFrom<A, ValueType>)
274 basic_array(A const &a) : lay(a.shape()), sto{lay.size(), mem::do_not_initialize} {
275 static_assert(std::is_constructible_v<ValueType, get_value_t<A>>, "Error in nda::basic_array: Incompatible value types in constructor");
276 if constexpr (std::is_trivial_v<ValueType> or is_complex_v<ValueType>) {
277 // trivial and complex value types can use the optimized assign_from_ndarray
278 if constexpr (std::is_same_v<ValueType, get_value_t<A>>)
279 assign_from_ndarray(a);
280 else
281 assign_from_ndarray(nda::map([](auto const &val) { return ValueType(val); })(a));
282 } else {
283 // general value types may not be default constructible -> use placement new
284 nda::for_each(lay.lengths(), [&](auto const &...is) { new (sto.data() + lay(is...)) ValueType{a(is...)}; });
285 }

306 // Get the corresponding shape from a nested initializer list.
307 template <typename L>
308 static auto shape_from_init_list(std::initializer_list<L> const &l) noexcept {
309 const auto [min, max] = std::minmax_element(std::begin(l), std::end(l), [](auto &&x, auto &&y) { return x.size() < y.size(); });
310 EXPECTS_WITH_MESSAGE(min->size() == max->size(),
311 "Error in nda::basic_array: Arrays can only be initialized with rectangular initializer lists");
312 return stdutil::front_append(shape_from_init_list(*max), long(l.size()));
313 }

443 template <std::integral Int = long>
444 static basic_array rand(std::array<Int, Rank> const &shape)
445 requires(std::is_floating_point_v<ValueType> or nda::is_complex_v<ValueType>)
446 {
447 auto static gen = std::mt19937_64{};
448 auto res = basic_array{shape};
449 if constexpr (nda::is_complex_v<ValueType>) {
450 auto static dist = std::uniform_real_distribution<typename ValueType::value_type>(0.0, 1.0);
451 using namespace std::complex_literals;
452 for (auto &x : res) x = dist(gen) + 1i * dist(gen);
453 } else {
454 auto static dist = std::uniform_real_distribution<ValueType>(0.0, 1.0);
455 for (auto &x : res) x = dist(gen);
456 }
457 return res;
458 }

757template <char ResultAlgebra, bool SelfIsRvalue, typename Self, typename... Ts>
758FORCEINLINE static decltype(auto) call(Self &&self, Ts const &...idxs) noexcept(has_no_boundcheck) {
759 // resulting value type
760 using r_v_t = std::conditional_t<std::is_const_v<std::remove_reference_t<Self>>, ValueType const, ValueType>;
761
762 // behavior depends on the given arguments
763 if constexpr (clef::is_any_lazy<Ts...>) {
764 // if there are lazy arguments, e.g. as in A(i_) << i_, a lazy expression is returned
765 return clef::make_expr_call(std::forward<Self>(self), idxs...);
766 } else if constexpr (sizeof...(Ts) == 0) {
767 // if no arguments are given, a full view is returned
768 return basic_array_view<r_v_t, Rank, LayoutPolicy, Algebra, AccessorPolicy, OwningPolicy>{self.lay, self.sto};
769 } else {
770 // otherwise we check the arguments and either access a single element or make a slice
771 static_assert(((layout_t::template argument_is_allowed_for_call_or_slice<Ts> + ...) > 0),
772 "Error in array/view: Slice arguments must be convertible to range, ellipsis, or long (or string if the layout permits it)");
773
774 // number of arguments convertible to long
775 static constexpr int n_args_long = (layout_t::template argument_is_allowed_for_call<Ts> + ...);
776
777 if constexpr (n_args_long == rank) {
778 // access a single element
779 long offset = self.lay(idxs...);
780 if constexpr (is_view or not SelfIsRvalue) {
781 // if the calling object is a view or an lvalue, we return a reference
782 return AccessorPolicy::template accessor<r_v_t>::access(self.sto.data(), offset);
783 } else {
784 // otherwise, we return a copy of the value
785 return ValueType{self.sto[offset]};
786 }
787 } else {
788 // access a slice of the view/array
789 auto const [offset, idxm] = self.lay.slice(idxs...);
790 static constexpr auto res_rank = decltype(idxm)::rank();
791 // resulting algebra
792 static constexpr char newAlgebra = (ResultAlgebra == 'M' and (res_rank == 1) ? 'V' : ResultAlgebra);
793 // resulting layout policy
794 using r_layout_p = typename detail::layout_to_policy<std::decay_t<decltype(idxm)>>::type;
795 return basic_array_view<r_v_t, res_rank, r_layout_p, newAlgebra, AccessorPolicy, OwningPolicy>{std::move(idxm), {self.sto, offset}};
796 }
797 }
798}

892// Make an iterator for the view/array depending on its type.
893template <typename Iterator>
894[[nodiscard]] auto make_iterator(bool at_end) const noexcept {
895 if constexpr (iterator_rank == Rank) {
896 // multi-dimensional iterator
897 if constexpr (layout_t::is_stride_order_C()) {
898 // C-order case (array_iterator already traverses the data in C-order)
899 return Iterator{indexmap().lengths(), indexmap().strides(), sto.data(), at_end};
900 } else {
901 // general case (we need to permute the shape and the strides according to the stride order of the layout)
902 return Iterator{nda::permutations::apply(layout_t::stride_order, indexmap().lengths()),
903 nda::permutations::apply(layout_t::stride_order, indexmap().strides()), sto.data(), at_end};
904 }
905 } else {
906 // 1-dimensional iterator
907 return Iterator{std::array<long, 1>{size()}, std::array<long, 1>{indexmap().min_stride()}, sto.data(), at_end};
908 }
909}

1029 static_assert(std::is_assignable_v<value_type &, get_value_t<RHS>>, "Error in assign_from_ndarray: Incompatible value types");
1030
1031 // are both operands nda::MemoryArray types?
1032 static constexpr bool both_in_memory = MemoryArray<self_t> and MemoryArray<RHS>;
1033
1034 // do both operands have the same stride order?
1035 static constexpr bool same_stride_order = get_layout_info<self_t>.stride_order == get_layout_info<RHS>.stride_order;
1036
1037 // compile-time check for device arrays to avoid runtime errors
1038 static_assert(!(mem::on_device<self_t> or mem::on_device<RHS>) or (both_in_memory and same_stride_order and have_same_value_type_v<self_t, RHS>),
1039 "Error in assign_from_ndarray: Assignment to/from device arrays is not supported for the given types.");
1040
1041 // prefer optimized options if possible
1042 if constexpr (both_in_memory and same_stride_order) {
1043 if (rhs.empty()) return;
1044 // are both operands strided in 1d?
1045 static constexpr bool both_1d_strided = has_layout_strided_1d<self_t> and has_layout_strided_1d<RHS>;
1046 if constexpr (mem::on_host<self_t, RHS> and both_1d_strided) {
1047 // vectorizable copy on host
1048 for (long i = 0; i < size(); ++i) (*this)(_linear_index_t{i}) = rhs(_linear_index_t{i});
1049 return;
1050 } else if constexpr (!mem::on_host<self_t, RHS> and have_same_value_type_v<self_t, RHS>) {
1051 // check for block-layout and use mem::memcpy2D if possible
1052 auto bl_layout_dst = get_block_layout(*this);
1053 auto bl_layout_src = get_block_layout(rhs);
1054 if (bl_layout_dst && bl_layout_src) {
1055 auto [n_bl_dst, bl_size_dst, bl_str_dst] = *bl_layout_dst;
1056 auto [n_bl_src, bl_size_src, bl_str_src] = *bl_layout_src;
1057 // check that the total memory size is the same
1058 if (n_bl_dst * bl_size_dst != n_bl_src * bl_size_src) NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Incompatible block sizes";
1059 // if either destination or source consists of a single block, we can chunk it up to make the layouts compatible
1060 if (n_bl_dst == 1 && n_bl_src > 1) {
1061 n_bl_dst = n_bl_src;
1062 bl_size_dst /= n_bl_src;
1063 bl_str_dst = bl_size_dst;
1064 }
1065 if (n_bl_src == 1 && n_bl_dst > 1) {
1066 n_bl_src = n_bl_dst;
1067 bl_size_src /= n_bl_dst;
1068 bl_str_src = bl_size_src;
1069 }
1070 // copy only if block-layouts are compatible, otherwise continue to fallback
1071 if (n_bl_dst == n_bl_src && bl_size_dst == bl_size_src) {
1072 mem::memcpy2D<mem::get_addr_space<self_t>, mem::get_addr_space<RHS>>((void *)data(), bl_str_dst * sizeof(value_type), (void *)rhs.data(),
1073 bl_str_src * sizeof(value_type), bl_size_src * sizeof(value_type),
1074 n_bl_src);
1075 return;
1076 }
1077 }
1078 }
1079 }
1080 // otherwise fallback to elementwise assignment
1081 if constexpr (mem::on_device<self_t> || mem::on_device<RHS>) {
1082 NDA_RUNTIME_ERROR << "Error in assign_from_ndarray: Fallback to elementwise assignment not implemented for arrays/views on the GPU";
1083 }
1084 nda::for_each(shape(), [this, &rhs](auto const &...args) { (*this)(args...) = rhs(args...); });
1085}

1087// Implementation to fill a view/array with a constant scalar value.
1088template <typename Scalar>
1089void fill_with_scalar(Scalar const &scalar) {
1090 if constexpr (mem::on_host<self_t>) {
1091 if constexpr (has_layout_strided_1d<self_t>) {
1092 const long L = size();
1093 auto *__restrict const p = data(); // no alias possible here!
1094 if constexpr (has_contiguous_layout<self_t>) {
1095 for (long i = 0; i < L; ++i) p[i] = scalar;
1096 } else {
1097 const long stri = indexmap().min_stride();
1098 const long Lstri = L * stri;
1099 for (long i = 0; i != Lstri; i += stri) p[i] = scalar;
1100 }
1101 } else {
1102 for (auto &x : *this) x = scalar;
1103 }
1104 } else if constexpr (mem::on_device<self_t> or mem::on_unified<self_t>) {
1105 if constexpr (has_layout_strided_1d<self_t>) {
1106 if constexpr (has_contiguous_layout<self_t>) {
1107 mem::fill_n<mem::get_addr_space<self_t>>(data(), size(), value_type(scalar));
1108 } else {
1109 const long stri = indexmap().min_stride();
1110 mem::fill2D_n<mem::get_addr_space<self_t>>(data(), stri, 1, size(), value_type(scalar));
1111 }
1112 } else {
1113 auto bl_layout = get_block_layout(*this);
1114 if (bl_layout) {
1115 auto [n_bl, bl_size, bl_str] = *bl_layout;
1116 mem::fill2D_n<mem::get_addr_space<self_t>>(data(), bl_str, bl_size, n_bl, value_type(scalar));
1117 } else {
1118 // MAM: implement recursive call to fill_with_scalar on (i,nda::ellipsis{})
1119 NDA_RUNTIME_ERROR << "fill_with_scalar: Not implemented on device for generic (non-blocked) layout.";
1120 }
1121 }
1122 }
1123}

1125// Implementation of the assignment from a scalar value.
1126template <typename Scalar>
1127void assign_from_scalar(Scalar const &scalar) noexcept {
1128 static_assert(!is_const, "Error in assign_from_ndarray: Cannot assign to a const view");
1129 if constexpr (Algebra != 'M') {
1130 // element-wise assignment for non-matrix algebras
1131 fill_with_scalar(scalar);
1132 } else {
1133 // a scalar has to be interpreted as a unit matrix for matrix algebras (the scalar in the shortest diagonal)
1134 // FIXME : A priori faster to put 0 everywhere and then change the diag to avoid the if.
1135 // FIXME : Benchmark and confirm.
1136 if constexpr (is_scalar_or_convertible_v<Scalar>)
1137 fill_with_scalar(0);
1138 else
1139 fill_with_scalar(Scalar{0 * scalar}); // FIXME : improve this
1140 diagonal(*this).fill_with_scalar(scalar);
1141 }
1142}
