55 requires((MemoryMatrix<A> or is_conj_array_expr<A>)
56 and have_same_value_type_v<A, X, Y> and mem::have_compatible_addr_space<A, X, Y> and is_blas_lapack_v<get_value_t<A>>)
57 void gemv(get_value_t<A> alpha, A const &a, X const &x, get_value_t<A> beta, Y &&y) { // NOLINT (temporary views are allowed here)
58 // get the underlying matrix in case it is given as a conjugate expression
59 auto &mat = get_array(a);
60
61 // check the dimensions of the input/output arrays/views
62 auto [m, n] = mat.shape();
63 EXPECTS(m == y.size());
64 EXPECTS(n == x.size());
65
66 // arrays/views must be BLAS compatible
67 EXPECTS(mat.indexmap().min_stride() == 1);
68
69 // We need to swap axis for the transpose case
70 if constexpr (has_C_layout<A>) std::swap(m, n);
71
72 // perform actual library call
73 if constexpr (mem::have_device_compatible_addr_space<A, X, Y>) {
74#if defined(NDA_HAVE_DEVICE)
75 device::gemv(get_op<A>, m, n, alpha, mat.data(), get_ld(mat), x.data(), x.indexmap().strides()[0], beta, y.data(), y.indexmap().strides()[0]);
76#else
77 compile_error_no_gpu();
78#endif
79 } else {
80 f77::gemv(get_op<A>, m, n, alpha, mat.data(), get_ld(mat), x.data(), x.indexmap().strides()[0], beta, y.data(), y.indexmap().strides()[0]);
81 }
82 }
