44 // Helper function to get the shape and total size of the gathered array/view.
45 template <typename A>
46 requires(is_regular_or_view_v<A> and std::decay_t<A>::is_stride_order_C())
47 auto mpi_gather_shape_impl(A const &a, mpi::communicator comm, int root, bool all) {
48 auto dims = a.shape();
49 dims[0] = mpi::all_reduce(dims[0], comm);
50 auto gathered_size = std::accumulate(dims.begin(), dims.end(), 1l, std::multiplies<>());
51 if (!all && comm.rank() != root) dims = nda::stdutil::make_initialized_array<dims.size()>(0l);
52 return std::make_pair(dims, gathered_size);
53 }

55 // Helper function that (all)gathers arrays/views.
56 template <typename A1, typename A2>
57 requires(is_regular_or_view_v<A1> and std::decay_t<A1>::is_stride_order_C()
58 and is_regular_or_view_v<A2> and std::decay_t<A2>::is_stride_order_C())
59 void mpi_gather_impl(A1 const &a_in, A2 &&a_out, mpi::communicator comm = {}, int root = 0, bool all = false) { // NOLINT
60 // check the shape of the input arrays/views
61 EXPECTS_WITH_MESSAGE(detail::have_mpi_equal_shapes(a_in(nda::range(1), nda::ellipsis{}), comm),
62 "Error in nda::detail::mpi_gather_impl: Shapes of arrays/views must be equal save the first one");
63
64 // simply copy if there is no active MPI environment or if the communicator size is < 2
65 if (not mpi::has_env || comm.size() < 2) {
66 a_out = a_in;
67 return;
68 }
69
70 // check if the input arrays/views can be used in the MPI call
71 detail::check_layout_mpi_compatible(a_in, "detail::mpi_gather_impl");
72
73 // get output shape, resize or check the output array/view and prepare output span
74 auto [dims, gathered_size] = mpi_gather_shape_impl(a_in, comm, root, all);
75 auto a_out_span = std::span{a_out.data(), 0};
76 if (all || (comm.rank() == root)) {
77 // check if the output array/view can be used in the MPI call
78 check_layout_mpi_compatible(a_out, "detail::mpi_gather_impl");
79
80 // resize/check the size of the output array/view
81 nda::resize_or_check_if_view(a_out, dims);
82
83 // prepare the output span
84 a_out_span = std::span{a_out.data(), static_cast<std::size_t>(a_out.size())};
85 }
86
87 // gather the data
88 auto a_in_span = std::span{a_in.data(), static_cast<std::size_t>(a_in.size())};
89 mpi::gather_range(a_in_span, a_out_span, gathered_size, comm, root, all);
90 }
