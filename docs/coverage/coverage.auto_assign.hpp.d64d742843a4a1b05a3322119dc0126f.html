82 /**
83 * @brief Assign values to the underlying object of a lazy function call expression.
84 *
85 * @details This calls the `clef_auto_assign` overload for the underlying object of the given expression using ADL.
86 * It has to be implemented for all supported types.
87 *
88 * @tparam F Type of the callable object in the function call expression.
89 * @tparam RHS Type of the right-hand side.
90 * @tparam Is Integer labels of the placeholders in the function call expression.
91 * @param ex nda::clef::expr object with the nda::clef::tags::function tag.
92 * @param rhs Right-hand side object.
93 */
94 template <typename F, typename RHS, int... Is>
95 FORCEINLINE void operator<<(expr<tags::function, F, placeholder<Is>...> const &ex, RHS &&rhs) {
96 static_assert(detail::all_different(Is...), "Error in clef operator<<: Two of the placeholders on the LHS are the same");
97 clef_auto_assign(std::get<0>(ex.childs), make_function(std::forward<RHS>(rhs), placeholder<Is>()...));
98 }
