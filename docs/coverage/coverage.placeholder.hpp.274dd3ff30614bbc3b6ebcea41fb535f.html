32 /**
33 * @brief A placeholder is an empty struct, labelled by an `int`.
34 *
35 * @details It is used in lazy expressions. For example:
36 *
37 * @include doc_clef_placeholder.cpp
38 *
39 * Output:
40 *
41 * ```
42 * 3
43 * ```
44 *
45 * Here `expr` is a lazy binary nda::clef::expr with the nda::clef::tags::plus tag, which can be evaluated later on
46 * with the nda::clef::eval function and by assigning values to the placeholders (see nda::clef::pair).
47 *
48 * @tparam N Integer label (must be < 64).
49 */
50 template <int N>
51 struct placeholder {
52 static_assert(N >= 0 && N < 64, "Placeholder index must be in {0, 1, ..., 63}");
53
54 /// Integer label.
55 static constexpr int index = N;
56
57 /**
58 * @brief Assign a value to the placeholder.
59 *
60 * @tparam RHS Type of the right-hand side.
61 * @param rhs Right-hand side of the assignment.
62 * @return An nda::clef::pair object, containing the integer label of the placeholder and the value assigned to it.
63 */
64 template <typename RHS>
65 200 pair<N, RHS> operator=(RHS &&rhs) const { // NOLINT (we want to return a pair)
66 200 return {std::forward<RHS>(rhs)};
67 }
68
69 /**
70 * @brief Function call operator.
71 *
72 * @tparam Args Types of the function call arguments.
73 * @param args Function call arguments.
74 * @return An nda::clef::expr object with the nda::clef::tags::function tag containing the placeholder (for the
75 * callable) and the given arguments.
76 */
77 template <typename... Args>
78 6 auto operator()(Args &&...args) const {
79 6 return expr<tags::function, placeholder, expr_storage_t<Args>...>{tags::function{}, *this, std::forward<Args>(args)...};
80 }
81
82 /**
83 * @brief Subscript operator.
84 *
85 * @tparam T Type of the subscript argument.
86 * @param t Subscript argument.
87 * @return An nda::clef::expr object with the nda::clef::tags::subscript tag containing the placeholder (for the
88 * object to be accessed) and the given argument.
89 */
90 template <typename T>
91 6 auto operator[](T &&t) const {
92 6 return expr<tags::subscript, placeholder, expr_storage_t<T>>{tags::subscript{}, *this, std::forward<T>(t)};
93 }
94 };

96 /**
97 * @brief A pair consisting of a placeholder and its assigned value.
98 *
99 * @details In most cases, the user should not have to explicitly create or handle pair objects. Instead, they are
100 * constructed indirectly by assigning a value to a placeholder when calling nda::clef::eval
101 * (see nda::clef::placeholder for an example).
102 *
103 * @tparam N Integer label of the placeholder.
104 * @tparam T Value type.
105 */
106 template <int N, typename T>
107 struct pair {
108 /// Value assigned to the placeholder (can be an lvalue reference).
109 T rhs;
110
111 /// Integer label of the placeholder.
112 static constexpr int p = N;
113
114 /// Type of the value after applying `std::decay`.
115 using value_type = std::decay_t<T>;
116 };
