32 /**
33 * @brief A placeholder is an empty struct, labelled by an int.
34 *
35 * @details It is used in lazy expressions. For example:
36 *
37 * @code{.cpp}
38 * nda::clef::placeholder<0> i_;
39 * nda::clef::placeholder<1> j_;
40 * auto expr = i_ + j_;
41 * auto res = nda::clef::eval(expr, i_ = 1.0, j_ = 2.0); // double res = 3.0;
42 * @endcode
43 *
44 * Here `expr` is a lazy binary nda::clef::expr with the nda::clef::tags::plus tag, which can be evaluated later on
45 * with the nda::clef::eval function and by assigning values to the placeholders (see nda::clef::pair).
46 *
47 * @tparam N Integer label (must be < 64).
48 */
49 template <int N>
50 struct placeholder {
51 static_assert(N >= 0 && N < 64, "Placeholder index must be in {0, 1, ..., 63}");
52
53 /// Integer label.
54 static constexpr int index = N;
55
56 /**
57 * @brief Assign a value to the placeholder.
58 *
59 * @tparam RHS Type of the right-hand side.
60 * @param rhs Right-hand side of the assignment.
61 * @return An nda::clef::pair object, containing the integer label of the placeholder and the value assigned to it.
62 */
63 template <typename RHS>
64 186 pair<N, RHS> operator=(RHS &&rhs) const { // NOLINT (we want to return a pair)
65 186 return {std::forward<RHS>(rhs)};
66 }
67
68 /**
69 * @brief Function call operator.
70 *
71 * @tparam Args Types of the function call arguments.
72 * @param args Function call arguments.
73 * @return An nda::clef::expr object with the nda::clef::tags::function tag containing the placeholder (for the
74 * callable) and the given arguments.
75 */
76 template <typename... Args>
77 6 auto operator()(Args &&...args) const {
78 6 return expr<tags::function, placeholder, expr_storage_t<Args>...>{tags::function{}, *this, std::forward<Args>(args)...};
79 }
80
81 /**
82 * @brief Subscript operator.
83 *
84 * @tparam T Type of the subscript argument.
85 * @param t Subscript argument.
86 * @return An nda::clef::expr object with the nda::clef::tags::subscript tag containing the placeholder (for the
87 * object to be accessed) and the given argument.
88 */
89 template <typename T>
90 auto operator[](T &&t) const {
91 return expr<tags::subscript, placeholder, expr_storage_t<T>>{tags::subscript{}, *this, std::forward<T>(t)};
92 }
93 };

95 /**
96 * @brief A pair consisting of a placeholder and its assigned value.
97 *
98 * @details In most cases, the user should not have to explicitly create or handle pair objects. Instead, they are
99 * constructed indirectly by assigning a value to a placeholder when calling nda::clef::eval
100 * (see nda::clef::placeholder for an example).
101 *
102 * @tparam N Integer label of the placeholder.
103 * @tparam T Value type.
104 */
105 template <int N, typename T>
106 struct pair {
107 /// Value assigned to the placeholder (can be an lvalue reference).
108 T rhs;
109
110 /// Integer label of the placeholder.
111 static constexpr int p = N;
112
113 /// Type of the value after applying std::decay.
114 using value_type = std::decay_t<T>;
115 };
