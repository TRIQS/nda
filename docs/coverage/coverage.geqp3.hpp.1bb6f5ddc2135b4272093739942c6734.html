30 /**
31 * @ingroup linalg_lapack
32 * @brief Interface to the LAPACK `%geqp3` routine.
33 *
34 * @details Computes a QR factorization of a matrix \f$ \mathbf{A} \f$ with column pivoting:
35 * \f[
36 * \mathbf{A P} = \mathbf{Q R}
37 * \f]
38 * using Level 3 BLAS.
39 *
40 * The matrix \f$ \mathbf{Q} \f$ is represented as a product of elementary reflectors
41 * \f[
42 * \mathbf{Q} = \mathbf{H}(1) \mathbf{H}(2) \ldots \mathbf{H}(k) \; ,
43 * \f]
44 * where \f$ k = \min(m,n) \f$.
45 *
46 * Each \f$ \mathbf{H}(i) \f$ has the form
47 * \f[
48 * \mathbf{H}(i) = \mathbf{I} - \tau_i * \mathbf{v}_i \mathbf{v}_i^H
49 * \f]
50 * where \f$ \tau_i \f$ is a real/complex scalar, and \f$ \mathbf{v}_i \f$ is a real/complex vector with
51 * - elements \f$ 1 \f$ to \f$ i - 1 \f$ equal to 0,
52 * - element \f$ i \f$ equal to 1 and
53 * - elements \f$ i + 1 \f$ to \f$ m \f$ stored on exit in the elements \f$ i + 1 \f$ to \f$ m \f$ in the column \f$ i
54 * \f$ of \f$ \mathbf{A} \f$.
55 *
56 * @note All input arrays are required to satisfy nda::mem::have_host_compatible_addr_space and \f$ \mathbf{A} \f$ is
57 * required to have nda::F_layout.
58 *
59 * @tparam A nda::blas_lapack::BlasArray<2> type.
60 * @tparam JPVT nda::blas_lapack::PivotArrayFor<A, 1> type.
61 * @tparam TAU nda::blas_lapack::BlasArrayFor<A, 1> type.
62 * @tparam W1 nda::blas_lapack::BlasArrayFor<A, 1> type.
63 * @tparam W2 nda::blas_lapack::BlasArrayRealFor<A, 1> type.
64 * @param a Input/output matrix. On entry, the \f$ m \times n \f$ matrix \f$ \mathbf{A} \f$. On exit, the upper
65 * triangle of the array contains the \f$ \min(m,n) \times n \f$ upper trapezoidal matrix \f$ \mathbf{R} \f$; the
66 * elements below the diagonal, together with the array \f$ \mathbf{\tau} \f$, represent the unitary matrix \f$
67 * \mathbf{Q} \f$ as a product of \f$ \min(m,n) \f$ elementary reflectors.
68 * @param jpvt Input/output vector. On entry, if the j<sup>th</sup> element is \f$ \neq 0 \f$, the j<sup>th</sup>
69 * column of \f$ \mathbf{A} \f$ is permuted to the front of \f$ \mathbf{A P} \f$ (a leading column); if the j<sup>th
70 * </sup> element is equal 0, the j<sup>th</sup> column of \f$ \mathbf{A} \f$ is a free column. On exit, if the
71 * j<sup>th</sup> element is equal \f$ k \f$, then the j<sup>th</sup> column of \f$ \mathbf{A P} \f$ was the
72 * k<sup>th</sup> column of \f$ \mathbf{A} \f$.
73 * @param tau Output vector. The scalar factors \f$ \tau_i \f$ of the elementary reflectors \f$ \mathbf{H}(i) \f$.
74 * @param work Ouput vector. Workspace array used by the LAPACK routine.
75 * @param rwork Ouput vector. Workspace array used by the LAPACK routine.
76 * @return Integer return code from the LAPACK call.
77 */
