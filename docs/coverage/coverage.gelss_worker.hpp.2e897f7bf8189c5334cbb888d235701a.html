42 /**
43 * @brief Worker class for solving linear least squares problems.
44 *
45 * @details Solving a linear least squares problem means finding the minimum norm solution \f$ \mathbf{x} \f$ of a
46 * linear system of equations, i.e.
47 * \f[
48 * \min_x | \mathbf{b} - \mathbf{A x} |_2 \; ,
49 * \f]
50 * where \f$ \mathbf{A} \f$ is a given matrix and \f$ \mathbf{b} \f$ is a given vector (although it can also be a
51 * matrix, in this case one searches for a solution matrix \f$ \mathbf{X} \f$).
52 *
53 * Let \f$ \mathbf{A} \in \mathbb{C}^{M \times N} \f$ with rank \f$ \rho \leq \min(M, N) \f$. Its singular value
54 * decomposition (SVD) is given by
55 * \f[
56 * \mathbf{A} = \mathbf{U} \mathbf{\Sigma} \mathbf{V}^H =
57 * \begin{bmatrix} \mathbf{U}_R & \mathbf{U}_N \end{bmatrix}
58 * \begin{bmatrix} \mathbf{S} & 0 \\ 0 & 0 \end{bmatrix}
59 * \begin{bmatrix} \mathbf{V}_R^H \\ \mathbf{V}_N^H \end{bmatrix} \; ,
60 * \f]
61 * where \f$ \mathbf{U}_R \in \mathbb{C}^{M \times \rho} \f$, \f$ \mathbf{U}_N \in \mathbb{C}^{M \times (M - \rho)}
62 * \f$, \f$ \mathbf{S} \in \mathbb{R}^{\rho \times \rho} \f$, \f$ \mathbf{V}_R \in \mathbb{C}^{N \times \rho} \f$,
63 * and \f$ \mathbf{V}_N \in \mathbb{C}^{N \times (N - \rho)} \f$.
64 *
65 * The least squares solution can then be written as
66 * \f[
67 * \mathbf{x} = \mathbf{A}^{+} \mathbf{b} = \mathbf{V} \mathbf{\Sigma}^{+} \mathbf{U}^H \mathbf{b} =
68 * \mathbf{V}_R \mathbf{S}^{-1} \mathbf{U}_R^H \mathbf{b} \; ,
69 * \f]
70 * where \f$ \mathbf{M}^{+} \f$ denotes the Moore-Penrose pseudo-inverse of the matrix \f$ \mathbf{M} \f$, and the
71 * residual error as
72 * \f[
73 * \epsilon = \left\| \mathbf{A} \mathbf{x} - \mathbf{b} \right\|_2 = \left\| \mathbf{U}_N^H \mathbf{b} \right\|_2
74 * \; .
75 * \f]
76 *
77 * See <a href="https://math.stackexchange.com/questions/772039/how-does-the-svd-solve-the-least-squares-problem">this
78 * question</a> on math.stackexchange for more information.
79 *
80 * @tparam T Value type of the given problem.
81 */

183 /**
184 * @brief Specialized worker class for solving linear least squares problems while enforcing a certain hermitian
185 * symmetry.
186 *
187 * @details This is a more specialized version of nda::lapack::gelss_worker that is tailored to perform tail fitting
188 * for scalar- or matrix-valued functions defined on a Matsubara frequency mesh satisfying the symmetry \f$ \left[
189 * f(-i\omega_n) \right]_{ij} = \left[ f(i\omega_n) \right]_{ji}^* \f$.
190 *
191 * The least squares problem to be solved is given by
192 * \f[
193 * \mathbf{A}_E \mathbf{X} = \begin{bmatrix} \mathbf{A} \\ \mathbf{A}^* \end{bmatrix} \mathbf{X} =
194 * \begin{bmatrix} \mathbf{B} \\ \tilde{\mathbf{B}} \end{bmatrix} = \mathbf{B}_E \; ,

197 * \mathbf{X} = \tilde{\mathbf{B}} \f$ to account for the symmetry given above.
198 *
199 * The calculated solution \f$ \mathbf{X} \f$ is explictly symmetrized at the end to ensure that the symmetry is
200 * satisfied.

203 */

236 if (not inner_matrix_dim.has_value())
237 NDA_RUNTIME_ERROR << "Error in nda::lapack::gelss_worker_hermitian: Inner matrix dimension required for hermitian least square fitting";

240 // take the inner 'adjoint' of a matrix C:
241 // * reshape C -> C': (M, N) -> (M, N', d, d)
242 // * for each m and n: C'(m, n, :, :) -> C'(m, n, :, :)^/dagger
243 // * reshape C' -> C: (M, N', d, d) -> (M, N)
244 auto inner_adjoint = [d](auto &C) {
245 NDA_ASSERT2(C.shape()[1] % (d * d) == 0, "Error in nda::lapack::gelss_worker_hermitian: Data shape incompatible with inner matrix dimension");
246 auto shape = C.shape();
247
248 // get extent N' of second dimension
249 long N = shape[1] / (d * d);
250
251 // reshape, transpose and take the complex conjugate
252 array<std::complex<double>, 4> arr_dag =
253 conj(permuted_indices_view<encode(std::array{0, 1, 3, 2})>(reshape(C, std::array{shape[0], N, d, d})));
254
255 // return the result in a new matrix
256 return matrix<std::complex<double>>{reshape(std::move(arr_dag), shape)};
257 };
258
259 // solve the extended system vstack(A, A*) * X = vstack(B, B_dag)
260 auto B_dag = inner_adjoint(B);
261 auto [x, err] = lss_herm_(vstack(B, B_dag));
262
263 // resymmetrize the results to cure small hermiticity violations
264 return std::pair<matrix<std::complex<double>>, double>{0.5 * (x + inner_adjoint(x)), err};
265 }

268 // Worker for the original least squares problem.
269 gelss_worker<std::complex<double>> lss_;
270
271 // Worker for the extended least squares problem.
272 gelss_worker<std::complex<double>> lss_herm_;
273 };
