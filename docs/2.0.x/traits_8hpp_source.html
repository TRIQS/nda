251 enum class layout_prop_e : uint64_t {
252 none = 0x0,
253 strided_1d = 0x1,
254 smallest_stride_is_one = 0x2,
255 contiguous = strided_1d | smallest_stride_is_one
256 };
257
266 inline constexpr bool layout_property_compatible(layout_prop_e from, layout_prop_e to) {
267 if (from == layout_prop_e::contiguous) return true;
268 return ((to == layout_prop_e::none) or (to == from));
269 }
270
278 constexpr layout_prop_e operator|(layout_prop_e lhs, layout_prop_e rhs) { return layout_prop_e(uint64_t(lhs) | uint64_t(rhs)); }
279
287 constexpr layout_prop_e operator&(layout_prop_e lhs, layout_prop_e rhs) { return layout_prop_e{uint64_t(lhs) & uint64_t(rhs)}; }

295 inline constexpr bool has_strided_1d(layout_prop_e lp) { return uint64_t(lp) & uint64_t(layout_prop_e::strided_1d); }
