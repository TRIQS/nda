55 void gemv(get_value_t<A> alpha, A const &a, X const &x, get_value_t<A> beta, Y &&y) { // NOLINT (temporary views are allowed here)
56 // get the underlying matrix in case it is given as a conjugate expression
57 auto &mat = get_array(a);
58
59 // check the dimensions of the input/output arrays/views
60 auto [m, n] = mat.shape();
61 EXPECTS(m == y.size());
62 EXPECTS(n == x.size());
63
64 // arrays/views must be BLAS compatible
